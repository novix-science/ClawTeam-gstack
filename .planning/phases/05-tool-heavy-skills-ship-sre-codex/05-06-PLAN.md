```python
"""/land-and-deploy handler (SKILL-15).

4-phase pipeline: precondition → CI wait → deploy → health probe → write deploy.md.
Never crashes on a failed step: every failure path writes deploy.md with
deploy_status in {failed, pending} so EvidenceGate can surface the outcome.
"""
from __future__ import annotations
import json
import re
import shlex
import subprocess
import time
import urllib.error
import urllib.request
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.plugins.skill_errors import SkillPreconditionError
from clawteam.spawn.invoke import invoke_native_cli
from clawteam.templates.gstack.schemas.deploy_notes import DeployNotes

_DEFAULT_CI_WAIT = 1800  # seconds (30 min)
_DEFAULT_DEPLOY_VERIFY = 120  # seconds

_URL_RE = re.compile(r"https?://[^\s]+")


def _read_ship_notes(sprint_dir: Path) -> dict[str, Any]:
    """Parse ship-notes.md frontmatter; raise SkillPreconditionError on missing/failed."""
    path = sprint_dir / "ship-notes.md"
    if not path.exists():
        raise SkillPreconditionError(
            skill="/land-and-deploy", role="shipper",
            message="Run /ship first (ship-notes.md not found)",
        )
    text = path.read_text()
    # Simple frontmatter parse — find ---...--- block at top
    if not text.startswith("---"):
        raise SkillPreconditionError(
            skill="/land-and-deploy", role="shipper",
            message="ship-notes.md has no frontmatter",
        )
    end = text.find("\n---", 3)
    if end == -1:
        raise SkillPreconditionError(
            skill="/land-and-deploy", role="shipper",
            message="ship-notes.md frontmatter unterminated",
        )
    fm_lines = text[3:end].splitlines()
    fm: dict[str, Any] = {}
    for line in fm_lines:
        if ":" not in line:
            continue
        k, _, v = line.partition(":")
        v = v.strip()
        # Strip surrounding quotes
        if v.startswith(("'", '"')) and v.endswith(("'", '"')) and len(v) >= 2:
            v = v[1:-1]
        # Boolean / null coercion
        if v == "true":
            fm[k.strip()] = True
        elif v == "false":
            fm[k.strip()] = False
        elif v == "null":
            fm[k.strip()] = None
        elif v.startswith("[") or v.startswith("{"):
            try:
                fm[k.strip()] = json.loads(v)
            except json.JSONDecodeError:
                fm[k.strip()] = v
        else:
            fm[k.strip()] = v
    if fm.get("ship_status") != "succeeded":
        raise SkillPreconditionError(
            skill="/land-and-deploy", role="shipper",
            message=(
                f"ship-notes.md has ship_status={fm.get('ship_status')!r}; "
                "Run /ship successfully first"
            ),
        )
    return fm


def _head_probe(url: str, timeout: float = 10.0) -> bool:
    """Returns True if HEAD request returns 2xx or 3xx; else False (no exceptions)."""
    try:
        req = urllib.request.Request(url, method="HEAD")
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            return 200 <= resp.status < 400
    except (urllib.error.URLError, urllib.error.HTTPError, TimeoutError, OSError):
        return False


def _wait_for_health(url: str, max_seconds: int) -> bool:
    """Exponential backoff probe up to max_seconds. Returns True on first 2xx/3xx."""
    if not url:
        return False
    deadline = time.monotonic() + max_seconds
    delay = 2.0
    while time.monotonic() < deadline:
        if _head_probe(url):
            return True
        time.sleep(min(delay, max(1.0, deadline - time.monotonic())))
        delay = min(delay * 1.5, 15.0)
    return False


def _extract_deploy_url(stdout: str) -> str:
    """Find the first https?:// URL in stdout."""
    match = _URL_RE.search(stdout)
    return match.group(0) if match else ""


def _build_deploy_cmd(provider: str, project: str, custom_cmd: str) -> list[str]:
    if provider == "vercel":
        return ["vercel", "deploy", "--prod", "--yes"]
    if provider == "netlify":
        return ["netlify", "deploy", "--prod"]
    if provider == "fly":
        return ["flyctl", "deploy", "--app", project]
    if provider == "custom":
        # Safe split — shlex posix mode, shell=False enforced at invoke layer
        return shlex.split(custom_cmd) if custom_cmd else []
    raise ValueError(f"unknown provider {provider!r}")


def _current_commit_sha(cwd: Path) -> str:
    """Read HEAD SHA via git; returns empty string on failure (best-effort)."""
    try:
        result = invoke_native_cli(
            ["git", "rev-parse", "HEAD"], cwd=cwd, timeout=10,
        )
        return result.stdout.strip()
    except Exception:
        return ""


def _emit_question_run_setup(sprint_dir: Path) -> None:
    """Emit a question artifact asking user to run /setup-deploy first."""
    q_dir = sprint_dir / "questions"
    q_dir.mkdir(parents=True, exist_ok=True)
    (q_dir / "001_run_setup_deploy.md").write_text(
        "---\nquestion_id: 001\npriority: blocking\n---\n\n"
        "### Run /setup-deploy\n\n"
        "gstack.toml has no [deploy] block. Please run /setup-deploy (sre role) "
        "to configure the deploy provider before re-running /land-and-deploy.\n"
    )


def _write_deploy_notes(
    sprint_dir: Path,
    *,
    sprint_id: str,
    deploy_status: str,
    deploy_url: str,
    provider: str,
    commit_sha: str,
) -> Path:
    schema = DeployNotes(
        artifact_type="deploy-notes",
        deploy_status=deploy_status,
        deploy_url=deploy_url or "pending",
        provider=provider if provider in ("vercel", "netlify", "fly", "custom") else "custom",
        deployed_at=datetime.now(timezone.utc).isoformat(),
        commit_sha=commit_sha or "0000000",
        sprint_id=sprint_id,
        created_at=datetime.now(timezone.utc).isoformat(),
    )
    d = schema.model_dump()
    lines = ["---"]
    for k, v in d.items():
        if isinstance(v, bool):
            lines.append(f"{k}: {str(v).lower()}")
        elif isinstance(v, str):
            lines.append(f"{k}: {v!r}")
        else:
            lines.append(f"{k}: {v}")
    lines.append("---")
    lines.append("")
    artifact_path = sprint_dir / "deploy.md"
    sprint_dir.mkdir(parents=True, exist_ok=True)
    atomic_write_text(artifact_path, "\n".join(lines) + "\n")
    return artifact_path


def land_and_deploy_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/land-and-deploy entry point."""
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    cwd = Path(getattr(ctx, "workspace_dir", sprint_dir))

    # 1. Precondition: ship-notes.md with ship_status=succeeded
    ship_fm = _read_ship_notes(sprint_dir)
    pr_url = ship_fm.get("pr_url", "")

    # 2. Determine deploy config
    tmpl = getattr(ctx, "template", None) or getattr(ctx, "tmpl", None)
    deploy_cfg = getattr(tmpl, "deploy", None) if tmpl is not None else None
    canary_cfg = getattr(tmpl, "canary", None) if tmpl is not None else None
    ci_timeout = (
        int(canary_cfg.ci_wait_timeout_seconds)
        if canary_cfg is not None else _DEFAULT_CI_WAIT
    )
    verify_timeout = int(args.get("deploy_verify_timeout_seconds", _DEFAULT_DEPLOY_VERIFY))
    commit_sha = _current_commit_sha(cwd)

    if deploy_cfg is None:
        _emit_question_run_setup(sprint_dir)
        artifact = _write_deploy_notes(
            sprint_dir, sprint_id=sprint_id, deploy_status="pending",
            deploy_url="pending", provider="custom", commit_sha=commit_sha,
        )
        return {
            "artifact_path": str(artifact), "deploy_status": "pending",
            "reason": "no_deploy_block",
        }

    # 3. CI wait via gh pr checks --watch
    if pr_url:
        try:
            ci_result = invoke_native_cli(
                ["gh", "pr", "checks", "--watch", pr_url],
                cwd=cwd, timeout=ci_timeout,
            )
            if ci_result.returncode != 0:
                artifact = _write_deploy_notes(
                    sprint_dir, sprint_id=sprint_id, deploy_status="failed",
                    deploy_url="pending", provider=deploy_cfg.provider,
                    commit_sha=commit_sha,
                )
                return {
                    "artifact_path": str(artifact), "deploy_status": "failed",
                    "failure_step": "ci", "exit_code": ci_result.returncode,
                }
        except subprocess.TimeoutExpired:
            artifact = _write_deploy_notes(
                sprint_dir, sprint_id=sprint_id, deploy_status="failed",
                deploy_url="pending", provider=deploy_cfg.provider,
                commit_sha=commit_sha,
            )
            return {
                "artifact_path": str(artifact), "deploy_status": "failed",
                "failure_step": "ci_timeout",
            }

    # 4. Deploy
    deploy_cmd = _build_deploy_cmd(
        deploy_cfg.provider, deploy_cfg.project, deploy_cfg.custom_deploy_cmd,
    )
    if not deploy_cmd:
        artifact = _write_deploy_notes(
            sprint_dir, sprint_id=sprint_id, deploy_status="failed",
            deploy_url="pending", provider=deploy_cfg.provider,
            commit_sha=commit_sha,
        )
        return {
            "artifact_path": str(artifact), "deploy_status": "failed",
            "failure_step": "deploy_cmd_empty",
        }
    deploy_result = invoke_native_cli(deploy_cmd, cwd=cwd, timeout=600)
    if deploy_result.returncode != 0:
        artifact = _write_deploy_notes(
            sprint_dir, sprint_id=sprint_id, deploy_status="failed",
            deploy_url="pending", provider=deploy_cfg.provider,
            commit_sha=commit_sha,
        )
        return {
            "artifact_path": str(artifact), "deploy_status": "failed",
            "failure_step": "deploy", "exit_code": deploy_result.returncode,
        }
    deploy_url = _extract_deploy_url(deploy_result.stdout)

    # 5. Health probe with exponential backoff
    healthy = _wait_for_health(deploy_url, verify_timeout)
    status = "succeeded" if healthy else "failed"
    artifact = _write_deploy_notes(
        sprint_dir, sprint_id=sprint_id, deploy_status=status,
        deploy_url=deploy_url or "pending", provider=deploy_cfg.provider,
        commit_sha=commit_sha,
    )
    return {
        "artifact_path": str(artifact), "deploy_status": status,
        "deploy_url": deploy_url,
    }
```
