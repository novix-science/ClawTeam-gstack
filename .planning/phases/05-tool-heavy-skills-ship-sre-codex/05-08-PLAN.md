```python
"""Canary HTTP poller + regression evaluator (SKILL-17)."""
from __future__ import annotations
import json
import time
import urllib.error
import urllib.request
from dataclasses import dataclass, field
from pathlib import Path
from typing import Callable

# Lazy import of get_data_dir to handle either clawteam.utils or clawteam.paths
try:
    from clawteam.utils import get_data_dir  # type: ignore
except ImportError:
    try:
        from clawteam.paths import get_data_dir  # type: ignore
    except ImportError:
        # Last-resort fallback
        def get_data_dir() -> Path:
            return Path.home() / ".clawteam"


@dataclass
class PollResult:
    http_2xx_count: int = 0
    http_5xx_count: int = 0
    response_times_ms: list[float] = field(default_factory=list)
    js_console_errors: list[str] = field(default_factory=list)

    @property
    def avg_response_ms(self) -> float:
        return (
            sum(self.response_times_ms) / len(self.response_times_ms)
            if self.response_times_ms else 0.0
        )

    @property
    def total_requests(self) -> int:
        return self.http_2xx_count + self.http_5xx_count


def _default_http_fn(url: str) -> tuple[int, float]:
    """Single HTTP GET; returns (status_code, elapsed_ms). 5xx on network error."""
    start = time.monotonic()
    try:
        req = urllib.request.Request(url, method="GET")
        with urllib.request.urlopen(req, timeout=10) as resp:
            status = resp.status
    except (urllib.error.URLError, urllib.error.HTTPError, TimeoutError, OSError):
        status = 599  # treat transport error as 5xx
    elapsed_ms = (time.monotonic() - start) * 1000
    return status, elapsed_ms


def poll_window(
    url: str,
    *,
    window_seconds: float,
    poll_interval_seconds: float,
    http_fn: Callable[[str], tuple[int, float]] | None = None,
    sleep_fn: Callable[[float], None] | None = None,
) -> PollResult:
    """Poll url every poll_interval_seconds for up to window_seconds.

    http_fn and sleep_fn are injectable for tests (avoid wall-clock dependence).
    """
    http_fn = http_fn or _default_http_fn
    sleep_fn = sleep_fn or time.sleep
    result = PollResult()
    deadline = time.monotonic() + window_seconds
    while time.monotonic() < deadline:
        status, elapsed_ms = http_fn(url)
        result.response_times_ms.append(elapsed_ms)
        if 200 <= status < 400:
            result.http_2xx_count += 1
        else:
            result.http_5xx_count += 1
        remaining = deadline - time.monotonic()
        if remaining > 0:
            sleep_fn(min(poll_interval_seconds, remaining))
    return result


def evaluate_regression(
    result: PollResult,
    *,
    baseline_avg_ms: float = 0.0,
    threshold_5xx_rate: float = 0.01,
    response_time_multiplier: float = 2.0,
) -> list[str]:
    """Return list of regression flag strings (empty if clean)."""
    flags: list[str] = []
    total = result.total_requests
    if total > 0:
        rate = result.http_5xx_count / total
        if rate > threshold_5xx_rate:
            flags.append("5xx_rate>1%")
    if baseline_avg_ms > 0 and result.avg_response_ms > baseline_avg_ms * response_time_multiplier:
        flags.append("response_time>2x_baseline")
    if result.js_console_errors:
        flags.append("js_console_error")
    return flags


def baseline_path(team: str, provider: str) -> Path:
    """Conventional location for /benchmark-written baseline JSON."""
    return get_data_dir() / "teams" / team / "baselines" / f"{provider}.json"


def load_baseline(team: str, provider: str) -> dict | None:
    """Best-effort baseline load; returns None on missing or malformed JSON."""
    path = baseline_path(team, provider)
    if not path.exists():
        return None
    try:
        return json.loads(path.read_text())
    except (json.JSONDecodeError, OSError):
        return None
```

```python
"""/canary handler (SKILL-17, D-09).

4-step: read deploy.md → poll window → evaluate regression → emit event + write report.
Browser mode (Playwright) is lazy-imported per Pitfall 5.
"""
from __future__ import annotations
import json
from datetime import datetime, timezone
from importlib.util import find_spec
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.events.types import DeployRegressionDetected
from clawteam.plugins.skill_errors import SkillPreconditionError
from clawteam.templates.gstack.schemas.canary_report import CanaryReport
from clawteam.templates.gstack.skills.canary.poller import (
    PollResult, poll_window, evaluate_regression, load_baseline,
)

_DEFAULT_WINDOW = 300
_DEFAULT_POLL_INTERVAL = 15


def _read_deploy_notes(sprint_dir: Path) -> dict[str, Any]:
    """Parse deploy.md frontmatter; raise SkillPreconditionError on missing."""
    path = sprint_dir / "deploy.md"
    if not path.exists():
        raise SkillPreconditionError(
            skill="/canary", role="sre",
            message="Run /land-and-deploy first (deploy.md not found)",
        )
    text = path.read_text()
    if not text.startswith("---"):
        raise SkillPreconditionError(
            skill="/canary", role="sre",
            message="deploy.md has no frontmatter",
        )
    end = text.find("\n---", 3)
    if end == -1:
        raise SkillPreconditionError(
            skill="/canary", role="sre",
            message="deploy.md frontmatter unterminated",
        )
    fm: dict[str, Any] = {}
    for line in text[3:end].splitlines():
        if ":" not in line:
            continue
        k, _, v = line.partition(":")
        v = v.strip()
        if v.startswith(("'", '"')) and v.endswith(("'", '"')) and len(v) >= 2:
            v = v[1:-1]
        fm[k.strip()] = v
    return fm


def _get_browser_errors(url: str) -> list[str]:
    """Lazy-import Playwright; scrape JS console errors.

    Returns [] if Playwright not installed or if browser invocation fails.
    Pitfall 5: Playwright is ONLY imported inside this function body.
    """
    if find_spec("playwright") is None:
        return []
    try:
        from playwright.sync_api import sync_playwright  # type: ignore
    except ImportError:
        return []
    errors: list[str] = []
    try:
        with sync_playwright() as p:
            browser = p.chromium.launch()
            page = browser.new_page()
            page.on("console", lambda msg: errors.append(msg.text)
                    if msg.type == "error" else None)
            page.goto(url, timeout=15000)
            page.wait_for_load_state("networkidle", timeout=10000)
            browser.close()
    except Exception as exc:  # noqa: BLE001 — browser errors never fail the handler
        errors.append(f"<browser-error>: {type(exc).__name__}: {exc}")
    return errors


def _write_canary_report(
    sprint_dir: Path,
    *,
    sprint_id: str,
    canary_status: str,
    result: PollResult,
    baseline_avg_ms: float,
    baseline_missing: bool,
    regression_flags: list[str],
) -> Path:
    schema = CanaryReport(
        artifact_type="canary-report",
        canary_status=canary_status,
        window_seconds=max(1, int(sum([1 for _ in range(max(1, result.total_requests))]))),
        http_2xx_count=result.http_2xx_count,
        http_5xx_count=result.http_5xx_count,
        avg_response_ms=result.avg_response_ms,
        pre_deploy_avg_response_ms=baseline_avg_ms,
        baseline_missing=baseline_missing,
        js_console_errors=result.js_console_errors,
        regression_flags=regression_flags,
        sprint_id=sprint_id,
        created_at=datetime.now(timezone.utc).isoformat(),
    )
    d = schema.model_dump()
    lines = ["---"]
    for k, v in d.items():
        if isinstance(v, bool):
            lines.append(f"{k}: {str(v).lower()}")
        elif isinstance(v, (list, dict)):
            lines.append(f"{k}: {json.dumps(v)}")
        elif isinstance(v, str):
            lines.append(f"{k}: {v!r}")
        else:
            lines.append(f"{k}: {v}")
    lines.append("---")
    lines.append("")
    artifact_path = sprint_dir / "canary-report.md"
    sprint_dir.mkdir(parents=True, exist_ok=True)
    atomic_write_text(artifact_path, "\n".join(lines) + "\n")
    return artifact_path


def canary_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/canary entry point (sre role)."""
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    team_name = getattr(ctx, "team_name", "default")

    deploy_fm = _read_deploy_notes(sprint_dir)
    deploy_url = deploy_fm.get("deploy_url", "")
    provider = deploy_fm.get("provider", "custom")

    tmpl = getattr(ctx, "template", None) or getattr(ctx, "tmpl", None)
    canary_cfg = getattr(tmpl, "canary", None) if tmpl is not None else None
    window = int(
        args.get("window_seconds",
                 canary_cfg.window_seconds if canary_cfg is not None else _DEFAULT_WINDOW)
    )
    interval = int(
        args.get("poll_interval_seconds",
                 canary_cfg.poll_interval_seconds if canary_cfg is not None else _DEFAULT_POLL_INTERVAL)
    )

    # Load baseline
    baseline = load_baseline(team_name, provider)
    baseline_missing = baseline is None
    baseline_avg_ms = float(baseline.get("avg_response_ms", 0.0)) if baseline else 0.0

    # Poll
    result = poll_window(
        deploy_url, window_seconds=window, poll_interval_seconds=interval,
    )

    # Browser mode (optional)
    if args.get("browser", False):
        result.js_console_errors.extend(_get_browser_errors(deploy_url))

    # Evaluate regression
    flags = evaluate_regression(result, baseline_avg_ms=baseline_avg_ms)
    canary_status = "regression" if flags else "clean"

    # Write report
    artifact = _write_canary_report(
        sprint_dir, sprint_id=sprint_id, canary_status=canary_status,
        result=result, baseline_avg_ms=baseline_avg_ms,
        baseline_missing=baseline_missing, regression_flags=flags,
    )

    # Emit event on regression
    if flags:
        bus = getattr(ctx, "bus", None)
        if bus is not None:
            bus.emit(DeployRegressionDetected(
                team_name=team_name,
                sprint_id=sprint_id,
                deploy_url=deploy_url,
                regression_flags=flags,
                http_2xx_count=result.http_2xx_count,
                http_5xx_count=result.http_5xx_count,
                avg_response_ms=result.avg_response_ms,
                pre_deploy_avg_response_ms=baseline_avg_ms,
            ))

    return {
        "artifact_path": str(artifact),
        "canary_status": canary_status,
        "regression_flags": flags,
    }
```
