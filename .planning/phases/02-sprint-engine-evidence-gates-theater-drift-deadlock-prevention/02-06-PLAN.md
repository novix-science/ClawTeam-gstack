    ```python
    """Artifact storage for structured handoff between harness phases."""

    from __future__ import annotations

    import json
    from datetime import datetime, timezone
    from pathlib import Path
    from typing import Any

    from clawteam.harness.errors import ArtifactTooLargeError


    def _now_iso() -> str:
        return datetime.now(timezone.utc).isoformat()


    class ArtifactStore:
        """File-based artifact storage for harness phases."""

        def __init__(
            self,
            base_dir: Path,
            team_name: str,
            harness_id: str,
            *,
            artifact_cap_bytes: int | None = None,
        ) -> None:
            self._dir = base_dir / team_name / harness_id / "artifacts"
            self._dir.mkdir(parents=True, exist_ok=True)
            self._cap = artifact_cap_bytes
            self._team_name = team_name  # exposed to BeforeFileWrite event

        def write(self, name: str, content: str, metadata: dict[str, Any] | None = None) -> Path:
            """Write an artifact file.

            Phase 2 hook chain (in order; short-circuit on first rejection per §02-RESEARCH §Integration Points):
              1. Size-cap check → raise ArtifactTooLargeError if len(content) > self._cap.
              2. Emit BeforeFileWrite(agent_name, path, size_bytes) → check event.veto.
                 → veto = True → raise FrozenPathError(event.veto_reason).
              3. Atomic write via path.write_text (existing behavior preserved).

            Note per §02-RESEARCH §Pitfall 8: frontmatter validation is NOT in this hook chain.
            The write-path stays BC for existing templates that write plain markdown without
            frontmatter. EvidenceGate (Plan 02-07) runs frontmatter validation at gate-check
            time, scoped to gstack artifacts only.
            """
            size_bytes = len(content.encode("utf-8"))

            # ── Hook 1: size cap ─────────────────────────────────────────────
            if self._cap is not None and size_bytes > self._cap:
                suggestion = (
                    f"split into {Path(name).stem}-part1{Path(name).suffix} etc. "
                    f"or raise cap via --artifact-cap"
                )
                raise ArtifactTooLargeError(
                    f"{name}: {size_bytes} bytes exceeds per-file cap of {self._cap} bytes. "
                    f"Suggestion: {suggestion}"
                )

            # ── Hook 2: BeforeFileWrite event → freeze / careful subscribers veto ──
            path = self._dir / name
            agent_name = ""
            if metadata:
                agent_name = str(metadata.get("agent", ""))
            try:
                from clawteam.events.global_bus import get_event_bus
                from clawteam.events.types import BeforeFileWrite
                from clawteam.harness.freeze_registry import FrozenPathError
            except Exception:  # pragma: no cover — defensive for partial-wave parallel execution
                BeforeFileWrite = None  # type: ignore[assignment]
                FrozenPathError = None  # type: ignore[assignment]

            if BeforeFileWrite is not None:
                event = BeforeFileWrite(
                    team_name=self._team_name,
                    agent_name=agent_name,
                    path=str(path),
                    size_bytes=size_bytes,
                )
                try:
                    get_event_bus().emit(event)
                except Exception:
                    # Never let a subscriber crash abort the write path; veto only.
                    event.veto = False
                if event.veto:
                    if FrozenPathError is not None:
                        raise FrozenPathError(
                            event.veto_reason or f"{path} is vetoed by safety-rail subscriber"
                        )
                    # Fallback when Plan 02-05 hasn't shipped yet — generic error.
                    raise ValueError(
                        event.veto_reason or f"{path} is vetoed by safety-rail subscriber"
                    )

            # ── Hook 3: atomic write (existing behavior preserved) ────────
            path.write_text(content, encoding="utf-8")
            if metadata:
                meta_path = self._dir / f"{name}.meta.json"
                meta_path.write_text(
                    json.dumps({**metadata, "written_at": _now_iso()}, indent=2),
                    encoding="utf-8",
                )
            return path

        def read(self, name: str) -> str | None:
            path = self._dir / name
            if path.is_file():
                return path.read_text(encoding="utf-8")
            return None

        def exists(self, name: str) -> bool:
            return (self._dir / name).is_file()

        def list_artifacts(self) -> list[dict[str, Any]]:
            result = []
            for path in sorted(self._dir.iterdir()):
                if path.name.endswith(".meta.json"):
                    continue
                entry: dict[str, Any] = {
                    "name": path.name,
                    "size": path.stat().st_size,
                }
                meta_path = self._dir / f"{path.name}.meta.json"
                if meta_path.is_file():
                    try:
                        entry["metadata"] = json.loads(meta_path.read_text(encoding="utf-8"))
                    except Exception:
                        pass
                result.append(entry)
            return result

        # ── Convenience methods for common artifact types ──

        def write_spec(self, content: str) -> Path:
            return self.write("spec.md", content, {"type": "specification", "phase": "plan"})

        def write_sprint_contract(self, contract_id: str, content: str) -> Path:
            return self.write(
                f"sprint-contract-{contract_id}.json",
                content,
                {"type": "sprint_contract", "phase": "plan"},
            )

        def write_evaluation(self, content: str) -> Path:
            return self.write("eval-report.json", content, {"type": "evaluation", "phase": "verify"})

        def write_ship_manifest(self, content: str) -> Path:
            return self.write("ship-manifest.json", content, {"type": "manifest", "phase": "ship"})
    ```
