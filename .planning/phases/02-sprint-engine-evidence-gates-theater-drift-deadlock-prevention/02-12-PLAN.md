- **Uniform JSON envelope per D-26** — every sub-command, success OR error, emits the
  same four-key shape: `{ok, data, warnings, error}`. MCP tools can parse without

  `MISSING_TEAM`.

  candidates)` → JSON error code `AMBIGUOUS_SPRINT` with the candidates list
  embedded in the human message.
- **BC invariant** — sprint sub-app is purely additive. Phase 0 regression matrix
  12/12 stays green because no existing CLI command semantics change.

    def most_recent_artifact(self, state: SprintState) -> str:
        """Return the most-recently-added artifact name; empty string when artifacts is empty.

        Insertion-order preserved by Python dict (3.7+); last key = most recent.
        """
        if not state.artifacts:
            return ""
        return list(state.artifacts.keys())[-1]

    def status_dict(self, state: SprintState) -> dict:
        """Shape consumed by `clawteam sprint status` (UX-03, D-24)."""
        return {
            "sprint_id": state.sprint_id,
            "team": state.team_name,
            "current_phase": state.current_phase,
            "status": state.status,
            "participants": list(state.participants),
            "pending_questions_count": len(state.pending_question_ids),
            "most_recent_artifact": self.most_recent_artifact(state),
            "auto_advance": state.auto_advance,
        }

    def show_dict(self, state: SprintState) -> dict:
        """Shape consumed by `clawteam sprint show` (UX-05, D-24).

        artifacts_list returns artifact NAMES only (not bodies) — keeps JSON bounded
        for UX readability. Full body fetch is out of scope for Phase 2 CLI; Phase 5
        may add `clawteam sprint artifact <sprint_id> <name>`.
        """
        return {
            "sprint_id": state.sprint_id,
            "team": state.team_name,
            "goal": state.goal,
            "current_phase": state.current_phase,
            "status": state.status,
            "participants": list(state.participants),
            "phase_history": list(state.phase_history),
            "artifacts_list": sorted(state.artifacts.keys()),
            "pending_question_ids": list(state.pending_question_ids),

            "created_at": state.created_at,

    """Uniform {ok, data, warnings, error} envelope for JSON output (D-26)."""

def _sprint_emit_err(code: str, message: str) -> None:
    """Uniform error envelope; always prints JSON (machine-readable even in human mode)."""
    payload = {"ok": False, "data": None, "warnings": [], "error": {"code": code, "message": message}}
    print(json.dumps(payload, indent=2, ensure_ascii=False))

def _render_sprint_human(data: dict) -> None:
    """Human-friendly rendering via rich (fallback to print for simple cases)."""
    try:
        from rich.console import Console
        console = Console()
        if "sprints" in data:  # list command

            console.print(f"  phase: {data['current_phase']} ({data['status']})")

def _resolve_sprint_or_err(team: str, id_or_prefix: str):
    """Shared resolve helper; translates AmbiguousSprintError / SprintNotFoundError → CLI err."""
    try:
        from clawteam.sprint.conductor import (
            SprintConductor, AmbiguousSprintError, SprintNotFoundError,
        )
    except ImportError as exc:
        _sprint_emit_err("IMPORT_FAILED", str(exc))
        return None, None
    c = SprintConductor(team_name=team)
    try:
        state = c.resolve_sprint(id_or_prefix)
        return c, state
    except AmbiguousSprintError as exc:
        _sprint_emit_err("AMBIGUOUS_SPRINT",
                         f"Ambiguous sprint prefix '{exc.prefix}'; candidates: {exc.candidates}")
    except SprintNotFoundError:
        _sprint_emit_err("SPRINT_NOT_FOUND", f"Sprint not found: {id_or_prefix}")
    return None, None
