    Step A — Create clawteam/harness/forced_progress_gate.py:

    ```python
    """forced_progress_gate — theater / no-progress detector (§02-CONTEXT D-14..D-17).

    Composed into PhaseRunner._gates[phase] AFTER EvidenceGate, BEFORE InteractionGate
    per §02-CONTEXT Integration Points. Reads SprintState.turn_counters and artifact-bytes
    delta to detect agents emitting zero-progress turns.

    Critical: per-agent isolation (Pitfall #16) — specialist agents (security, designer)
    waiting for engineer/reviewer are NOT theater. Only the agent with 2+ consecutive
    no-progress turns gets flagged.
    """

    from __future__ import annotations

    import uuid
    from datetime import datetime, timezone
    from pathlib import Path

    from clawteam.events.global_bus import get_event_bus
    from clawteam.events.types import ForcedProgressTriggered
    from clawteam.harness.phases import PhaseGate
    from clawteam.sprint.qa import Choice, Question
    from clawteam.team.models import get_data_dir


    NO_PROGRESS_THRESHOLD = 2  # §02-CONTEXT D-16


    class forced_progress_gate(PhaseGate):  # noqa: N801 — gate name matches §02-CONTEXT D-17 spelling
        """Detect 2+ consecutive no-progress turns per agent; write a multi-choice question on trigger."""

        def __init__(self) -> None:
            # Per-agent "last counter snapshot" — records the turn_counter value at the
            # last time we saw progress for this agent. If turn_counters[agent] - last_seen
            # > NO_PROGRESS_THRESHOLD AND no progress signal since, trigger.
            self._last_progress_snapshot: dict[str, int] = {}
            self._last_artifact_sizes: dict[str, int] = {}
            self._triggered_agents: set[str] = set()
            self._task_completed_agents: set[str] = set()  # populated by event subscriber

        def register(self) -> None:
            """Subscribe to TaskCompleted so we can reset counters when tasks complete."""
            from clawteam.events.types import TaskCompleted
            get_event_bus().subscribe(TaskCompleted, self._on_task_completed)

        def _on_task_completed(self, event) -> None:
            self._task_completed_agents.add(event.owner or "")

        def check(self, state) -> tuple[bool, str]:
            # Only apply to SprintState (not PhaseState); fall through otherwise.
            turn_counters = getattr(state, "turn_counters", None)
            if turn_counters is None:
                return True, ""

            # D-16 + Pitfall #16: STRICT per-agent attribution. NO sprint-total fallback.
            # Plan 02-06 Task 2 emits BeforeFileWrite with `metadata["agent"]`; ArtifactStore.write
            # records that on the per-artifact `.meta.json`. We compute per-agent byte totals by
            # scanning `<sprint_dir>/artifacts/*.meta.json` and summing only entries whose
            # `agent` field matches the agent under consideration. One agent's write MUST NOT
            # reset another agent's no-progress counter.
            per_agent_bytes = _compute_per_agent_artifact_bytes(state)

            triggered: list[str] = []
            for agent, counter in turn_counters.items():
                last_snapshot = self._last_progress_snapshot.get(agent, 0)
                no_progress_turns = counter - last_snapshot
                # TaskCompleted reset (own-agent only — event.owner matched at subscribe time)
                if agent in self._task_completed_agents:
                    self._last_progress_snapshot[agent] = counter
                    self._task_completed_agents.discard(agent)
                    continue
                # Per-agent artifact-bytes delta — strictly this agent's writes (agent authorship
                # recorded on .meta.json by Plan 02-06). Another agent's write does not count here.
                this_agent_bytes = per_agent_bytes.get(agent, 0)
                last_bytes = self._last_artifact_sizes.get(agent, 0)
                if this_agent_bytes > last_bytes:
                    self._last_progress_snapshot[agent] = counter
                    self._last_artifact_sizes[agent] = this_agent_bytes
                    continue
                if no_progress_turns >= NO_PROGRESS_THRESHOLD and agent not in self._triggered_agents:
                    triggered.append(agent)

            if not triggered:
                return True, ""

            # Write a question file for the FIRST triggered agent (one at a time;
            # subsequent triggers queue via repeat check()s).
            agent = triggered[0]
            question_id = self._write_question(state, agent)
            self._triggered_agents.add(agent)
            try:
                get_event_bus().emit(ForcedProgressTriggered(
                    team_name=getattr(state, "team", ""),
                    agent=agent,
                    consecutive_no_progress=turn_counters[agent] - self._last_progress_snapshot.get(agent, 0),
                    question_id=question_id,
                ))
            except Exception:
                pass
            return False, (
                f"forced_progress_gate triggered: agent {agent!r} has 2 consecutive "
                f"no-progress turns; question {question_id} written "
                f"(answer lifts suppression)."
            )

        def _write_question(self, state, agent: str) -> str:
            sprint_dir = (
                get_data_dir() / "teams" / state.team / "sprints" / state.sprint_id
            )
            questions_dir = sprint_dir / "questions"
            questions_dir.mkdir(parents=True, exist_ok=True)
            q_id = uuid.uuid4().hex[:8]
            question = Question(
                id=q_id,
                slug=f"forced-progress-{agent}",
                type="multi-choice",
                created_at=datetime.now(timezone.utc).isoformat(),
                sprint_id=state.sprint_id,
                phase=state.current_phase,
                author="forced_progress_gate",
                choices=[
                    Choice(id="A", label="keep waiting — agent will ship next turn"),
                    Choice(id="B", label="restart agent from clean state"),
                    Choice(id="C", label="abort sprint"),
                ],
            )
            body = (
                f"Agent `{agent}` has produced 2 consecutive no-progress turns "
                f"in phase `{state.current_phase}`. Net artifact bytes: no delta.\n\n"
                f"Pick an action:\n"
            )
            (questions_dir / f"{q_id}.md").write_text(
                question.to_markdown(body=body),
                encoding="utf-8",
            )
            return q_id

        def clear_agent_trigger(self, agent: str) -> None:
            """Called by Plan 02-11 Conductor when a matching answer file arrives —
            resets the per-agent trigger so the gate passes next time."""
            self._triggered_agents.discard(agent)
            # Counter snapshot refresh happens at check-time via progress signals.


    def _compute_per_agent_artifact_bytes(state) -> dict[str, int]:
        """Scan `<sprint_dir>/artifacts/*.meta.json` and sum bytes per author-agent.

        D-16 + Pitfall #16: NO sprint-total fallback. Each artifact counts ONLY
        toward the agent whose `metadata["agent"]` is recorded on its sibling
        `.meta.json` (written by Plan 02-06 Task 2). Artifacts without an
        author-agent recorded are ignored for progress attribution. This preserves
        per-agent isolation: one agent's write never clears another agent's
        no-progress counter.
        """
        import json
        from collections import defaultdict
        from pathlib import Path
        from clawteam.team.models import get_data_dir

        per_agent: defaultdict[str, int] = defaultdict(int)
        try:
            sprint_dir = (
                get_data_dir() / "teams" / state.team_name / "sprints" / state.sprint_id
            )
        except Exception:
            return dict(per_agent)
        artifacts_dir = sprint_dir / "artifacts"
        if not artifacts_dir.exists():
            return dict(per_agent)
        for meta_path in artifacts_dir.glob("*.meta.json"):
            try:
                meta = json.loads(meta_path.read_text(encoding="utf-8"))
            except Exception:
                continue
            agent = str(meta.get("agent") or "")
            size = int(meta.get("size_bytes") or 0)
            if not agent or size <= 0:
                continue
            per_agent[agent] += size
        return dict(per_agent)
    ```

    _seen_turn_ids: dict[str, set[str]] = defaultdict(set)
