```python
"""/learn handler (MEM-04, D-07, D-08).

Role-agnostic — any GSTACK_ROLES agent can dispatch. Side-effect is
on TeamMemoryStore under the current team name (resolved from ctx).

High-impact gating + conflict detection integration is instrumented in
Plan 06-11 (wraps store.write at the plugin boundary via a branch).
"""
from __future__ import annotations

from pathlib import Path
from typing import Any

from clawteam.memory import MemoryEntry, TeamMemoryStore
from clawteam.memory.search import search as memory_search


_VALID_ACTIONS = frozenset({"write", "list", "search", "prune"})


def _resolve_team_name(ctx: Any) -> str:
    return getattr(ctx, "team_name", "") or "default"


def _build_entry(store: TeamMemoryStore, *, role: str, args: dict[str, Any]) -> MemoryEntry:
    scope = args.get("scope", "team")
    entry_role = args.get("role") if scope == "role" else None
    author = args.get("author") or f"{role}@learn"
    title = args["title"]
    body = args.get("body", "")
    tags_raw = args.get("tags") or []
    if isinstance(tags_raw, str):
        tags = [t.strip() for t in tags_raw.split(",") if t.strip()]
    else:
        tags = list(tags_raw)
    impact = args.get("impact", "medium")
    if impact not in ("low", "medium", "high"):
        raise ValueError(f"invalid impact {impact!r}")
    if impact == "high" and "impact:high" not in tags:
        tags.append("impact:high")
    evidence = args.get("evidence", "") or ""
    confidence = float(args.get("confidence", 0.7))
    learned_from = args.get("learned_from", "artifact")
    sprint_id = args.get("sprint_id", "")
    phase = args.get("phase", "")

    entry_id = store.gen_id(author=author, title=title, tags=tags)
    return MemoryEntry(
        id=entry_id,
        author=author,
        sprint_id=sprint_id,
        phase=phase,
        title=title,
        body=body,
        tags=tags,
        evidence=evidence,
        confidence=confidence,
        learned_from=learned_from,
        scope=scope,
        role=entry_role,
    )


def _do_write(store: TeamMemoryStore, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    entry = _build_entry(store, role=role, args=args)
    entry_id = store.write(entry)
    return {
        "status": "written",
        "id": entry_id,
        "evidence_flagged": not bool(entry.evidence),
        "high_impact": "impact:high" in entry.tags,
    }


def _do_list(store: TeamMemoryStore, *, args: dict[str, Any]) -> dict[str, Any]:
    scope = args.get("scope", "team")
    tag = args.get("tag")
    role_arg = args.get("role") if scope == "role" else None
    entries = store.list(scope=scope, tag=tag, role=role_arg)
    return {"status": "listed", "entries": [e.model_dump() for e in entries]}


def _do_search(store: TeamMemoryStore, *, args: dict[str, Any]) -> dict[str, Any]:
    query = args.get("query", "")
    scope = args.get("scope", "team")
    tag = args.get("tag")
    role_arg = args.get("role") if scope == "role" else None
    results = memory_search(store, query, scope=scope, tag=tag, role=role_arg)
    explain = bool(args.get("explain"))
    payload = []
    for r in results:
        item = {
            "id": r.entry.id,
            "title": r.entry.title,
            "tags": r.entry.tags,
            "score": round(r.score, 4),
        }
        if explain:
            item.update({
                "recency": round(r.recency, 4),
                "provenance": round(r.provenance, 4),
                "decay": round(r.decay, 4),
            })
        payload.append(item)
    return {"status": "searched", "query": query, "results": payload}


def _do_prune(store: TeamMemoryStore, *, args: dict[str, Any]) -> dict[str, Any]:
    entry_id = args["id"]
    scope = args.get("scope", "team")
    role_arg = args.get("role") if scope == "role" else None
    store.prune(entry_id, scope=scope, role=role_arg)
    return {"status": "pruned", "id": entry_id}


def learn_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """Entry point invoked via SkillDispatcher. Role-agnostic (D-07)."""
    action = args.get("action", "")
    if action not in _VALID_ACTIONS:
        raise ValueError(f"unknown action {action!r}")

    team_name = _resolve_team_name(ctx)
    store = TeamMemoryStore(team_name)

    if action == "write":
        return _do_write(store, role=role, args=args)
    if action == "list":
        return _do_list(store, args=args)
    if action == "search":
        return _do_search(store, args=args)
    if action == "prune":
        return _do_prune(store, args=args)
    raise ValueError(f"unreachable action {action!r}")


__all__ = ["learn_handler"]
```

# Phase 6 Plan 06-10: clawteam learn ... subcommand group (MEM-04, D-08)
learn_app = typer.Typer(help="Team memory operations (/learn skill surface)")
app.add_typer(learn_app, name="learn")


def _learn_ctx(team_name: str):
    """Synthesize a minimal ctx object for the learn handler."""
    from types import SimpleNamespace
    return SimpleNamespace(team_name=team_name)


@learn_app.command("write")
def learn_write(
    team: str = typer.Option(..., "--team", help="Team name"),
    scope: str = typer.Option("team", "--scope", help="team | role"),
    role: Optional[str] = typer.Option(None, "--role", help="Required when scope=role"),
    title: str = typer.Option(..., "--title"),
    tags: str = typer.Option("", "--tags", help="Comma-separated"),
    evidence: str = typer.Option("", "--evidence", help="path:line | artifact ref"),
    confidence: float = typer.Option(0.7, "--confidence", min=0.0, max=1.0),
    learned_from: str = typer.Option("artifact", "--learned-from"),
    impact: str = typer.Option("medium", "--impact", help="low|medium|high"),
    sprint_id: str = typer.Option("", "--sprint-id"),
    body: str = typer.Argument(""),
):
    """Write a memory entry."""
    from clawteam.templates.gstack.skills.learn.handler import learn_handler
    args = {
        "action": "write", "scope": scope, "role": role, "title": title,
        "tags": tags, "evidence": evidence, "confidence": confidence,
        "learned_from": learned_from, "impact": impact, "sprint_id": sprint_id,
        "body": body,
    }
    try:
        result = learn_handler(_learn_ctx(team), role="cli", args=args)
    except (ValueError, Exception) as exc:
        _output({"error": str(exc)}, None)
        raise typer.Exit(code=1)
    _output(result, lambda d: print(f"Wrote {d['id']}"))


@learn_app.command("list")
def learn_list(
    team: str = typer.Option(..., "--team"),
    scope: str = typer.Option("team", "--scope"),
    role: Optional[str] = typer.Option(None, "--role"),
    tag: Optional[str] = typer.Option(None, "--tag"),
):
    """List memory entries under a scope."""
    from clawteam.templates.gstack.skills.learn.handler import learn_handler
    args = {"action": "list", "scope": scope, "role": role, "tag": tag}
    result = learn_handler(_learn_ctx(team), role="cli", args=args)
    _output(result, lambda d: _print_entries_table(d["entries"]))


def _print_entries_table(entries: list[dict]) -> None:
    from rich.table import Table
    tbl = Table(title="Memory entries")
    tbl.add_column("id"); tbl.add_column("tags"); tbl.add_column("date"); tbl.add_column("conf"); tbl.add_column("evd")
    for e in entries:
        tbl.add_row(
            e["id"], ",".join(e["tags"]), e["timestamp"][:10],
            f"{e['confidence']:.2f}", "yes" if e["evidence"] else "no",
        )
    console.print(tbl)


@learn_app.command("search")
def learn_search(
    query: str = typer.Argument(""),
    team: str = typer.Option(..., "--team"),
    scope: str = typer.Option("team", "--scope"),
    role: Optional[str] = typer.Option(None, "--role"),
    tag: Optional[str] = typer.Option(None, "--tag"),
    explain: bool = typer.Option(False, "--explain"),
):
    """Search memory entries by keyword + tag; ranked by recency × provenance × decay."""
    from clawteam.templates.gstack.skills.learn.handler import learn_handler
    args = {"action": "search", "query": query, "scope": scope, "role": role,
            "tag": tag, "explain": explain}
    result = learn_handler(_learn_ctx(team), role="cli", args=args)
    _output(result, lambda d: _print_search_table(d["results"], explain))


def _print_search_table(results: list[dict], explain: bool) -> None:
    from rich.table import Table
    tbl = Table(title="Search results")
    tbl.add_column("score"); tbl.add_column("id"); tbl.add_column("title")
    if explain:
        tbl.add_column("recency"); tbl.add_column("provenance"); tbl.add_column("decay")
    for r in results:
        row = [f"{r['score']:.3f}", r["id"], r["title"]]
        if explain:
            row.extend([f"{r['recency']:.2f}", f"{r['provenance']:.2f}", f"{r['decay']:.2f}"])
        tbl.add_row(*row)
    console.print(tbl)


@learn_app.command("prune")
def learn_prune(
    entry_id: str = typer.Argument(...),
    team: str = typer.Option(..., "--team"),
    scope: str = typer.Option("team", "--scope"),
    role: Optional[str] = typer.Option(None, "--role"),
):
    """Prune an entry (append tombstone; never delete history)."""
    from clawteam.templates.gstack.skills.learn.handler import learn_handler
    args = {"action": "prune", "id": entry_id, "scope": scope, "role": role}
    result = learn_handler(_learn_ctx(team), role="cli", args=args)
    _output(result, lambda d: print(f"Pruned {d['id']}"))
