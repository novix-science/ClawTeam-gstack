```python
"""Skill contribution type — Phase 5 substrate.

Plugins contribute skills via HarnessPlugin.contribute_skills() returning
a list[SkillRegistration]. PluginManager.get_plugin_skills() aggregates;
SkillDispatcher.dispatch() looks up + role-gates + tool-availability-checks
+ invokes the handler.
"""
from __future__ import annotations
from dataclasses import dataclass
from typing import Any, Callable


@dataclass(frozen=True)
class SkillRegistration:
    """One skill contribution from a HarnessPlugin.

    - name: slash-command name as agents invoke it, e.g. "/codex".
    - roles: frozenset of agent roles permitted to invoke; SkillNotPermitted
             raised on dispatch if role not in roles.
    - handler: callable with signature (ctx, *, role: str, args: dict) -> Any.
    - tool_available: optional callable () -> bool. If provided and returns
                      False at dispatch time, SkillUnavailable is raised with
                      install_hint before handler runs.
    - install_hint: text shown in SkillUnavailable (matches clawteam doctor).
    """
    name: str
    roles: frozenset[str]
    handler: Callable[..., Any]
    tool_available: Callable[[], bool] | None = None
    install_hint: str = ""
```

```python
"""Structured skill-dispatch errors (Phase 5 substrate, D-04)."""
from __future__ import annotations


class SkillError(Exception):
    """Base for all skill-dispatch failures (agent-visible, structured)."""

    def __init__(self, skill: str, role: str = "", message: str = "") -> None:
        self.skill = skill
        self.role = role
        self.message = message
        super().__init__(f"[{skill}] (role={role}) {message}")


class SkillUnavailable(SkillError):
    """Raised when a required external tool is missing (per D-04)."""

    def __init__(
        self,
        skill: str,
        binary: str,
        install_hint: str,
        role: str = "",
    ) -> None:
        self.binary = binary
        self.install_hint = install_hint
        super().__init__(
            skill=skill,
            role=role,
            message=f"tool '{binary}' not available. Install: {install_hint}",
        )


class SkillNotPermitted(SkillError):
    """Raised when an agent role invokes a skill not in its SkillRegistration.roles."""


class SkillPreconditionError(SkillError):
    """Raised when a skill's invariant isn't met (e.g., /land-and-deploy before /ship)."""
```

**3. `clawteam/plugins/skill_dispatcher.py` (NEW FILE, ~70 LOC):**

```python
"""Synchronous skill dispatch entry point (Phase 5 substrate).

Open Question 1 (RESEARCH.md): The agent-to-dispatcher wire-up (MCP tool call
or TurnEnvelope slash-command parser) is deferred to Phase 7. For Phase 5
tests, dispatch() is called directly.
"""
from __future__ import annotations
from typing import Any, Mapping

from clawteam.plugins.skill_errors import (
    SkillError,
    SkillNotPermitted,
    SkillUnavailable,
)
from clawteam.plugins.skill_registration import SkillRegistration


class SkillDispatcher:
    """Looks up a SkillRegistration by name, checks role + tool, invokes handler."""

    def __init__(self, registrations: Mapping[str, SkillRegistration]) -> None:
        self._registrations = dict(registrations)

    def dispatch(
        self,
        ctx: Any,
        *,
        skill_name: str,
        role: str,
        args: dict[str, Any] | None = None,
    ) -> Any:
        reg = self._registrations.get(skill_name)
        if reg is None:
            raise SkillError(
                skill=skill_name,
                role=role,
                message=f"unknown skill {skill_name!r}",
            )
        if role not in reg.roles:
            raise SkillNotPermitted(
                skill=skill_name,
                role=role,
                message=(
                    f"role '{role}' not in permitted roles "
                    f"{sorted(reg.roles)}"
                ),
            )
        if reg.tool_available is not None and not reg.tool_available():
            # install_hint from registration; binary name = skill trim "/"
            raise SkillUnavailable(
                skill=skill_name,
                binary=skill_name.lstrip("/"),
                install_hint=reg.install_hint,
                role=role,
            )
        return reg.handler(ctx, role=role, args=dict(args or {}))
```

**1. `clawteam/spawn/invoke.py` (NEW FILE, ~50 LOC):**

Thin wrapper around NativeCliAdapter.prepare_command + subprocess.run with scrub_env default and shell=False invariant (see 05-RESEARCH.md §A1 Option A, which planner ADOPTS per assumption A7-ASSUMED).

```python
"""Native CLI invocation wrapper (Phase 5 substrate, 05-RESEARCH §A1).

The NativeCliAdapter at clawteam/spawn/adapters.py only exposes
prepare_command() — it does NOT run subprocess or scrub env. This module
provides the missing invoke_native_cli() wrapper so Phase 5 skill handlers
have ONE call site with uniform env scrubbing, shell=False, and timeout.
"""
from __future__ import annotations
import os
import subprocess
from pathlib import Path
from typing import Mapping, Sequence

from clawteam.secrets import scrub_env
from clawteam.spawn.adapters import NativeCliAdapter

_ADAPTER = NativeCliAdapter()


def invoke_native_cli(
    command: Sequence[str],
    *,
    cwd: str | Path | None = None,
    prompt: str | None = None,
    timeout: float = 120.0,
    env: Mapping[str, str] | None = None,
    skip_permissions: bool = False,
    interactive: bool = False,
) -> subprocess.CompletedProcess[str]:
    """Prepare a command via NativeCliAdapter and run it with safe defaults.

    - shell=False is HARD-CODED (D-15 adversarial-input protection).
    - env defaults to scrub_env(os.environ) — caller can override by passing
      env=<explicit dict>; the wrapper then does NOT scrub the caller's dict.
    - TimeoutExpired / CalledProcessError / FileNotFoundError re-raise as-is.
    """
    prepared = _ADAPTER.prepare_command(
        command=list(command),
        prompt=prompt,
        cwd=cwd,
        skip_permissions=skip_permissions,
        interactive=interactive,
    )
    effective_env = dict(env) if env is not None else scrub_env(os.environ)
    return subprocess.run(
        prepared.final_command,
        cwd=str(cwd) if cwd is not None else None,
        env=effective_env,
        capture_output=True,
        text=True,
        timeout=timeout,
        check=False,
        shell=False,
    )
```
