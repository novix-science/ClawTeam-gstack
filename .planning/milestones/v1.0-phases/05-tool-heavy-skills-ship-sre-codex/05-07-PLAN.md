"""Doc file walker + code-reference extractor (SKILL-16).

Pure helpers — no git, no filesystem side effects beyond read.
"""
from __future__ import annotations
import re
from pathlib import Path

# Inline code blocks (backtick) — captures path-like or identifier-like content
_INLINE_CODE_RE = re.compile(r"`([^`\n]+)`")
# Path heuristics: contains "/" or ends in known extension
_PATH_LIKE_RE = re.compile(r"^[A-Za-z0-9_./\-]+(?:\.py|\.ts|\.js|\.tsx|\.jsx|\.md|\.toml|\.json|\.rs|\.go)$")
# Identifier heuristics: alphanumeric + underscore, starts with letter/underscore
_IDENT_RE = re.compile(r"^[A-Za-z_][A-Za-z0-9_]{1,}$")
# Signature patterns: 'def X', 'function X', 'class X', 'export function X'
_SIGNATURE_RE = re.compile(
    r"(?:export\s+)?(?:function|def|class)\s+([A-Za-z_][A-Za-z0-9_]*)"
)

_EXCLUDED_DIRS = frozenset({"node_modules", ".venv", ".git", "__pycache__", "dist", "build"})


def walk_docs(root: Path) -> list[Path]:
    """Return sorted list of .md files under root, excluding common build dirs."""
    results: list[Path] = []
    for path in root.rglob("*.md"):
        if any(part in _EXCLUDED_DIRS for part in path.parts):
            continue
        results.append(path)
    return sorted(results)


def extract_code_refs(md_content: str) -> set[str]:
    """Extract path-like and identifier-like tokens from inline code spans + signatures."""
    refs: set[str] = set()
    for match in _INLINE_CODE_RE.finditer(md_content):
        token = match.group(1).strip()
        if _PATH_LIKE_RE.fullmatch(token):
            refs.add(token)
        elif _IDENT_RE.fullmatch(token):
            refs.add(token)
    for match in _SIGNATURE_RE.finditer(md_content):
        refs.add(match.group(1))
    return refs
```

"""Unified-diff patch emitter for stale doc references (SKILL-16)."""
from __future__ import annotations
import difflib
from pathlib import Path


def emit_patches(
    doc_path: Path,
    stale_refs: set[str],
    *,
    max_patches: int = 50,
) -> str:
    """Emit a unified-diff-format patch string for stale_refs in doc_path.

    Returns empty string if stale_refs is empty. If more than max_patches
    stale refs, emits max_patches patches + a trailer note with count deferred.
    """
    if not stale_refs:
        return ""
    original_text = doc_path.read_text()
    original_lines = original_text.splitlines(keepends=True)

    # Simple strategy: for each stale_ref, comment out or flag the matching line.
    deferred = max(0, len(stale_refs) - max_patches)
    sorted_refs = sorted(stale_refs)[:max_patches]

    new_lines = list(original_lines)
    changed = False
    for ref in sorted_refs:
        for idx, line in enumerate(new_lines):
            if ref in line and not line.lstrip().startswith("<!-- STALE:"):
                # Prepend a STALE comment marker; leave content intact for review
                new_lines[idx] = f"<!-- STALE: `{ref}` was removed/renamed in this PR -->\n" + line
                changed = True
                break

    if not changed:
        return ""

    diff = "".join(
        difflib.unified_diff(
            original_lines,
            new_lines,
            fromfile=str(doc_path),
            tofile=str(doc_path) + ".proposed",
            n=2,
        )
    )

    if deferred:
        diff += (
            f"\n# NOTE: {deferred} additional stale refs deferred; "
            "re-run /document-release to continue.\n"
        )
    return diff
```

```python
"""/document-release handler (SKILL-16)."""
from __future__ import annotations
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.spawn.invoke import invoke_native_cli
from clawteam.templates.gstack.skills.document_release.doc_walker import (
    walk_docs, extract_code_refs,
)
from clawteam.templates.gstack.skills.document_release.patch_emitter import (
    emit_patches,
)

_DEFAULT_MAX_PATCHES_PER_RUN = 50


def _removed_paths_from_diff(cwd: Path, base: str, head: str) -> set[str]:
    """Return set of paths REMOVED in diff (status=D or missing from tree at head)."""
    # Use --diff-filter=D to get only deleted files; cheaper than full diff.
    result = invoke_native_cli(
        ["git", "diff", "--diff-filter=D", "--name-only", f"{base}..{head}"],
        cwd=cwd, timeout=30,
    )
    if result.returncode != 0:
        return set()
    return {line.strip() for line in result.stdout.splitlines() if line.strip()}


def _find_stale_refs(doc_path: Path, removed_paths: set[str]) -> set[str]:
    """Intersection of code refs in doc_path with removed_paths."""
    refs = extract_code_refs(doc_path.read_text())
    return refs & removed_paths


def document_release_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/document-release entry point. Idempotent + non-destructive (D-11)."""
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    cwd = Path(getattr(ctx, "workspace_dir", sprint_dir))
    base = args.get("base", "main")
    head = args.get("head", "HEAD")
    max_patches = int(args.get("max_patches_per_run", _DEFAULT_MAX_PATCHES_PER_RUN))

    removed = _removed_paths_from_diff(cwd, base, head)

    docs = walk_docs(cwd)

    patch_dir = sprint_dir / "docs-updates"
    patch_dir.mkdir(parents=True, exist_ok=True)

    stale_hits: list[tuple[Path, set[str], str]] = []  # (doc, refs, patch_str)
    total_patches_emitted = 0

    for doc in docs:
        doc_stale = _find_stale_refs(doc, removed)
        if not doc_stale:
            continue
        patch = emit_patches(doc, doc_stale, max_patches=max_patches)
        if not patch:
            continue
        stale_hits.append((doc, doc_stale, patch))
        total_patches_emitted += 1

    # Write patches under docs-updates/<ISO-8601>.diff
    timestamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ")
    if stale_hits:
        combined = "\n\n# ---\n\n".join(patch for _, _, patch in stale_hits)
        patch_file = patch_dir / f"{timestamp}.diff"
        atomic_write_text(patch_file, combined)

        # D-15 + Pitfall 7: emit question artifact only for non-trivial writes
        questions_dir = sprint_dir / "questions"
        questions_dir.mkdir(parents=True, exist_ok=True)
        for i, (doc, _, patch) in enumerate(stale_hits, start=1):
            if len(patch.splitlines()) > 5:
                q_path = questions_dir / f"docrel_{timestamp}_{i:03d}.md"
                q_path.write_text(
                    f"---\nquestion_id: docrel_{i:03d}\n"
                    f"priority: normal\n---\n\n"
                    f"### Apply patch to `{doc}`?\n\n"
                    f"Patch proposal at `docs-updates/{timestamp}.diff`:\n\n"
                    f"```\n{patch[:2000]}\n```\n\n"
                    "- yes: apply\n- no: skip\n"
                )

    # Summary artifact
    summary_path = sprint_dir / "document-release-summary.md"
    if not stale_hits:
        summary = (
            "---\nartifact_type: document-release-summary\n"
            f"created_at: {datetime.now(timezone.utc).isoformat()}\n"
            f"status: no_changes\npatches_emitted: 0\n---\n\n"
            "No stale references found; docs are in sync with diff.\n"
        )
    else:
        summary = (
            "---\nartifact_type: document-release-summary\n"
            f"created_at: {datetime.now(timezone.utc).isoformat()}\n"
            f"status: found_stale\npatches_emitted: {total_patches_emitted}\n"
            f"patch_file: docs-updates/{timestamp}.diff\n---\n\n"
            f"Found {total_patches_emitted} doc file(s) with stale references. "
            "Patches written non-destructively; apply after review.\n"
        )
    atomic_write_text(summary_path, summary)

    return {
        "status": "ok",
        "patches_emitted": total_patches_emitted,
        "summary_path": str(summary_path),
    }
```
