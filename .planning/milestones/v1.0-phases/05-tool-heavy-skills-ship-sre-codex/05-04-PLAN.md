- `open_pr(cwd, title, body) -> StepResult` — check `shutil.which("gh")` first; raise SkillUnavailable if missing; call `gh pr create --title <title> --body <body>`; parse last non-empty stdout line as PR URL (gh pr create prints URL to stdout).

```python
import json
import shutil
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from clawteam.plugins.skill_errors import SkillUnavailable
from clawteam.spawn.invoke import invoke_native_cli


@dataclass
class StepResult:
    success: bool
    details: dict[str, Any] = field(default_factory=dict)


_SMOKE_PY_TEMPLATE = '''"""Bootstrap smoke test — auto-generated by /ship (D-06).

TODO: flesh out this smoke test with real assertions before re-running /ship.
"""


def test_smoke():
    # Trivial truthy check — replace with real import + assertion.
    assert True
'''

_SMOKE_TS_TEMPLATE = '''// Bootstrap smoke test — auto-generated by /ship (D-06).
// TODO: flesh out this smoke test with real assertions before re-running /ship.
import { test, expect } from "vitest";

test("smoke test", () => {
  expect(true).toBe(true);
});
'''


def _detect_language(cwd: Path) -> str:
    if (cwd / "pyproject.toml").exists() or (cwd / "setup.py").exists():
        return "python"
    if (cwd / "package.json").exists():
        return "node"
    if (cwd / "go.mod").exists():
        return "go"
    if (cwd / "Cargo.toml").exists():
        return "rust"
    return "unknown"


def _has_tests(cwd: Path, language: str) -> bool:
    if language == "python":
        return any(cwd.glob("tests/test_*.py")) or any(cwd.glob("test_*.py"))
    if language == "node":
        return (
            (cwd / "tests").is_dir()
            or (cwd / "__tests__").is_dir()
            or any(cwd.glob("*.test.ts"))
            or any(cwd.glob("**/*.test.ts"))
        )
    if language == "go":
        return any(cwd.glob("**/*_test.go"))
    if language == "rust":
        return (cwd / "tests").is_dir()
    return False


def _bootstrap_tests(cwd: Path, language: str, sprint_dir: Path) -> None:
    tests_dir = cwd / "tests"
    tests_dir.mkdir(exist_ok=True)
    if language == "python":
        (tests_dir / "test_smoke.py").write_text(_SMOKE_PY_TEMPLATE)
    elif language == "node":
        (tests_dir / "smoke.test.ts").write_text(_SMOKE_TS_TEMPLATE)

    # Emit question artifact (D-06 human nudge)
    questions_dir = sprint_dir / "questions"
    questions_dir.mkdir(parents=True, exist_ok=True)
    (questions_dir / "001_flesh_out_smoke_test.md").write_text(
        "---\n"
        "question_id: 001\n"
        "priority: blocking\n"
        "---\n\n"
        "### Flesh out bootstrapped smoke test\n\n"
        "No test suite detected on this branch. /ship wrote a TODO smoke test at "
        "`tests/test_smoke.py` (python) or `tests/smoke.test.ts` (node).\n\n"
        "Please replace the trivial assertion with a real import + check that proves "
        "the project entry point loads. Then re-run /ship.\n"
    )


def sync_main(cwd: Path, branch: str, timeout: int = 120) -> StepResult:
    fetch = invoke_native_cli(["git", "fetch", "origin", "main"], cwd=cwd, timeout=timeout)
    if fetch.returncode != 0:
        return StepResult(False, {"error": "fetch_failed", "stderr": fetch.stderr})
    merge = invoke_native_cli(
        ["git", "merge", "origin/main", "--no-edit"], cwd=cwd, timeout=timeout,
    )
    if merge.returncode != 0:
        return StepResult(False, {"conflict": True, "stderr": merge.stderr})
    return StepResult(True, {"synced_onto": "main"})


def run_tests(cwd: Path, sprint_dir: Path, timeout: int = 600) -> StepResult:
    language = _detect_language(cwd)
    if language == "unknown":
        return StepResult(False, {"error": "unknown_language"})
    if not _has_tests(cwd, language):
        _bootstrap_tests(cwd, language, sprint_dir)
        return StepResult(
            False,
            {"bootstrapped": True, "language": language,
             "hint": "question emitted; fill smoke test and re-run /ship"},
        )
    # Run tests
    if language == "python":
        cmd = ["pytest", "-x", "--tb=short"]
    elif language == "node":
        cmd = ["npx", "vitest", "run"]  # TODO: jest detection; acceptable fallback
    elif language == "go":
        cmd = ["go", "test", "./..."]
    elif language == "rust":
        cmd = ["cargo", "test"]
    else:
        return StepResult(False, {"error": "unknown_language"})
    result = invoke_native_cli(cmd, cwd=cwd, timeout=timeout)
    return StepResult(
        result.returncode == 0,
        {"language": language, "exit_code": result.returncode},
    )


def audit_coverage(cwd: Path, threshold: float, timeout: int = 600) -> StepResult:
    language = _detect_language(cwd)
    if language == "python":
        # pytest-cov reports JSON to coverage.json when --cov-report=json given
        result = invoke_native_cli(
            ["pytest", "--cov", "--cov-report=json", "--cov-report=term", "--tb=no", "-q"],
            cwd=cwd, timeout=timeout,
        )
        cov_file = cwd / "coverage.json"
        if not cov_file.exists():
            return StepResult(False, {"error": "coverage_file_missing"})
        try:
            data = json.loads(cov_file.read_text())
            percent = float(data.get("totals", {}).get("percent_covered", 0.0)) / 100.0
        except (ValueError, KeyError) as exc:
            return StepResult(False, {"error": f"parse_failed: {exc}"})
        return StepResult(
            percent >= threshold,
            {"coverage": round(percent, 3), "threshold": threshold, "language": "python"},
        )
    elif language == "node":
        result = invoke_native_cli(
            ["npx", "c8", "--reporter=json-summary", "npx", "vitest", "run"],
            cwd=cwd, timeout=timeout,
        )
        cov_file = cwd / "coverage" / "coverage-summary.json"
        if not cov_file.exists():
            return StepResult(False, {"error": "coverage_file_missing"})
        try:
            data = json.loads(cov_file.read_text())
            percent = float(data["total"]["lines"]["pct"]) / 100.0
        except (ValueError, KeyError) as exc:
            return StepResult(False, {"error": f"parse_failed: {exc}"})
        return StepResult(
            percent >= threshold,
            {"coverage": round(percent, 3), "threshold": threshold, "language": "node"},
        )
    else:
        # go/rust coverage parsing is language-specific; skip deep integration for v1.
        return StepResult(
            True,
            {"coverage": None, "threshold": threshold, "language": language,
             "note": "coverage audit skipped for non-python/node"},
        )


def push(cwd: Path, branch: str, timeout: int = 120) -> StepResult:
    result = invoke_native_cli(
        ["git", "push", "-u", "origin", branch], cwd=cwd, timeout=timeout,
    )
    return StepResult(
        result.returncode == 0,
        {"branch": branch, "exit_code": result.returncode, "stderr": result.stderr},
    )


def open_pr(cwd: Path, title: str, body: str, timeout: int = 60) -> StepResult:
    if shutil.which("gh") is None:
        raise SkillUnavailable(
            skill="/ship",
            binary="gh",
            install_hint="apt install gh | brew install gh | winget install GitHub.cli",
        )
    result = invoke_native_cli(
        ["gh", "pr", "create", "--title", title, "--body", body],
        cwd=cwd, timeout=timeout,
    )
    if result.returncode != 0:
        return StepResult(False, {"error": "gh_failed", "stderr": result.stderr})
    # gh pr create prints the PR URL; take the last non-empty line
    pr_url = ""
    for line in reversed(result.stdout.splitlines()):
        line = line.strip()
        if line.startswith("http"):
            pr_url = line
            break
    return StepResult(
        bool(pr_url),
        {"pr_url": pr_url or None, "stdout": result.stdout[:500]},
    )
```

```python
    """Top-level tool_available for /ship skill registration."""
    return shutil.which("gh") is not None


def _resolve_threshold(ctx: Any) -> float:
    """Read coverage_threshold from tmpl.ship.coverage_threshold, else default."""
    tmpl = getattr(ctx, "template", None) or getattr(ctx, "tmpl", None)
    if tmpl is not None:
        ship_cfg = getattr(tmpl, "ship", None)
        if ship_cfg is not None and getattr(ship_cfg, "coverage_threshold", None) is not None:
            return float(ship_cfg.coverage_threshold)
    return _DEFAULT_COVERAGE_THRESHOLD


def _write_ship_notes(
    sprint_dir: Path,
    *,
    sprint_id: str,
    ship_status: str,
    steps_completed: list[str],
    coverage: float | None,
    coverage_threshold: float | None,
    failure_step: str | None,
    failure_reason: str | None,
    pr_url: str | None,
    branch: str | None,
    auto_invoked_skills: list[str],
) -> Path:
    created_at = datetime.now(timezone.utc).isoformat()
    schema = ShipNotes(
        artifact_type="ship-notes",
        sprint_id=sprint_id,
        created_at=created_at,
        persona="shipper",
        step_label="ship",
        done=(ship_status == "succeeded"),
        ship_status=ship_status,
        steps_completed=steps_completed,
        coverage=coverage,
        coverage_threshold=coverage_threshold,
        failure_step=failure_step,
        failure_reason=failure_reason,
        pr_url=pr_url,
        branch=branch,
        auto_invoked_skills=auto_invoked_skills,
    )
    # Hand-rolled YAML frontmatter emitter (same pattern as codex handler)
    d = schema.model_dump()
    lines = ["---"]
    for key, val in d.items():
        if isinstance(val, bool):
            lines.append(f"{key}: {str(val).lower()}")
        elif isinstance(val, (list, dict)):
            import json as _json
            lines.append(f"{key}: {_json.dumps(val)}")
        elif val is None:
            lines.append(f"{key}: null")
        elif isinstance(val, str):
            lines.append(f"{key}: {val!r}")
        else:
            lines.append(f"{key}: {val}")
    lines.append("---")
    lines.append("")
    sprint_dir.mkdir(parents=True, exist_ok=True)
    artifact_path = sprint_dir / "ship-notes.md"
    atomic_write_text(artifact_path, "\n".join(lines) + "\n")
    return artifact_path


def ship_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/ship entry point (SKILL-14, shipper role only)."""
    if not gh_available():
        raise SkillUnavailable(
            skill="/ship", binary="gh", install_hint=_GH_INSTALL_HINT, role=role,
        )

    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    cwd = Path(getattr(ctx, "workspace_dir", sprint_dir))
    branch = args.get("branch") or getattr(ctx, "branch", "HEAD")
    title = args.get("title", f"Ship: {sprint_id}")
    body = args.get("body", "")

    steps_completed: list[str] = []
    pr_url: str | None = None
    coverage: float | None = None
    coverage_threshold = _resolve_threshold(ctx)

    # Step 1: sync_main
    sync_result = sync_main(cwd, branch)
    if not sync_result.success:
        artifact = _write_ship_notes(
            sprint_dir, sprint_id=sprint_id, ship_status="failed",
            steps_completed=steps_completed, coverage=None,
            coverage_threshold=coverage_threshold, failure_step="sync",
            failure_reason=str(sync_result.details),
            pr_url=None, branch=branch, auto_invoked_skills=[],
        )
        return {"artifact_path": str(artifact), "ship_status": "failed", "failure_step": "sync"}
    steps_completed.append("sync")

    # Step 2: run_tests
    tests_result = run_tests(cwd, sprint_dir)
    if not tests_result.success:
        artifact = _write_ship_notes(
            sprint_dir, sprint_id=sprint_id, ship_status="failed",
            steps_completed=steps_completed, coverage=None,
            coverage_threshold=coverage_threshold, failure_step="test",
            failure_reason=str(tests_result.details),
            pr_url=None, branch=branch, auto_invoked_skills=[],
        )
        return {"artifact_path": str(artifact), "ship_status": "failed", "failure_step": "test"}
    steps_completed.append("test")

    # Step 3: audit_coverage
    cov_result = audit_coverage(cwd, coverage_threshold)
    coverage = cov_result.details.get("coverage")
    if not cov_result.success:
        artifact = _write_ship_notes(
            sprint_dir, sprint_id=sprint_id, ship_status="failed",
            steps_completed=steps_completed, coverage=coverage,
            coverage_threshold=coverage_threshold, failure_step="coverage",
            failure_reason=str(cov_result.details),
            pr_url=None, branch=branch, auto_invoked_skills=[],
        )
        return {"artifact_path": str(artifact), "ship_status": "failed", "failure_step": "coverage"}
    steps_completed.append("coverage")

    # Step 4: push
    push_result = push(cwd, branch)
    if not push_result.success:
        artifact = _write_ship_notes(
            sprint_dir, sprint_id=sprint_id, ship_status="failed",
            steps_completed=steps_completed, coverage=coverage,
            coverage_threshold=coverage_threshold, failure_step="push",
            failure_reason=str(push_result.details),
            pr_url=None, branch=branch, auto_invoked_skills=[],
        )
        return {"artifact_path": str(artifact), "ship_status": "failed", "failure_step": "push"}
    steps_completed.append("push")

    # Step 5: open_pr
    pr_result = open_pr(cwd, title, body)
    if not pr_result.success:
        artifact = _write_ship_notes(
            sprint_dir, sprint_id=sprint_id, ship_status="failed",
            steps_completed=steps_completed, coverage=coverage,
            coverage_threshold=coverage_threshold, failure_step="pr",
            failure_reason=str(pr_result.details),
            pr_url=None, branch=branch, auto_invoked_skills=[],
        )
        return {"artifact_path": str(artifact), "ship_status": "failed", "failure_step": "pr"}
    pr_url = pr_result.details.get("pr_url")
    steps_completed.append("pr")

    # All 5 steps succeeded
    artifact = _write_ship_notes(
        sprint_dir, sprint_id=sprint_id, ship_status="succeeded",
        steps_completed=steps_completed, coverage=coverage,
        coverage_threshold=coverage_threshold, failure_step=None,
        failure_reason=None, pr_url=pr_url, branch=branch,
        auto_invoked_skills=[],  # Plan 05-07 appends /document-release
    )
    return {
        "artifact_path": str(artifact),
        "ship_status": "succeeded",
        "pr_url": pr_url,
        "coverage": coverage,
    }
```
