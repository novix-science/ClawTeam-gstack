```python
"""/codex skill handler (SKILL-13, D-01 sub-package shape).

All external CLI invocation goes through invoke_native_cli (Wave 0 wrapper) so
env scrubbing + shell=False + timeout are enforced uniformly.
"""
from __future__ import annotations
import shutil
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Literal

from clawteam.fileutil import atomic_write_text
from clawteam.plugins.skill_errors import SkillUnavailable
from clawteam.spawn.invoke import invoke_native_cli
from clawteam.templates.gstack.schemas.codex_review import CodexReview

CodexMode = Literal["review", "adversarial", "consultation"]
_VALID_MODES = frozenset({"review", "adversarial", "consultation"})
_INSTALL_HINT = "npm install -g @openai/codex"
_CODEX_TIMEOUT = 600  # 10 min — codex can be slow on long files


def tool_available() -> bool:
    """Return True if `codex` CLI is on PATH."""
    return shutil.which("codex") is not None


def invoke_codex(
    prompt: str,
    *,
    mode: CodexMode,
    cwd: str | Path,
    timeout: int = _CODEX_TIMEOUT,
) -> str:
    """Invoke the codex CLI via invoke_native_cli; returns stdout.

    Raises SkillUnavailable if codex not on PATH.
    """
    if not tool_available():
        raise SkillUnavailable(
            skill="/codex",
            binary="codex",
            install_hint=_INSTALL_HINT,
        )
    result = invoke_native_cli(
        command=["codex", "exec"],
        prompt=prompt,
        cwd=cwd,
        timeout=timeout,
        skip_permissions=True,
        interactive=False,
    )
    return result.stdout


def _frontmatter_yaml(schema: CodexReview) -> str:
    """Serialize a CodexReview as YAML frontmatter + empty body.

    Uses a hand-rolled emitter (no pyyaml dep) matching Phase 3 schemas pattern.
    """
    d = schema.model_dump()
    lines = ["---"]
    for key in ("artifact_type", "mode", "target", "verdict", "summary",
               "sprint_id", "created_at", "persona", "step_label", "done"):
        val = d[key]
        if isinstance(val, bool):
            lines.append(f"{key}: {str(val).lower()}")
        elif isinstance(val, str):
            lines.append(f"{key}: {val!r}")
        else:
            lines.append(f"{key}: {val}")
    lines.append("---")
    lines.append("")
    return "\n".join(lines) + "\n"


def codex_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """Entry point invoked by SkillDispatcher.

    args keys:
        mode: "review" | "adversarial" | "consultation"  (required)
        target: short label — file path, diff range, or prompt subject (required)
        prompt: full text sent to codex exec (required)

    ctx MUST expose:
        sprint_dir: Path  — directory to write artifact
        sprint_id: str

    Returns: {"artifact_path": str, "mode": str, "stdout": str}
    """
    mode = args.get("mode")
    target = args.get("target", "")
    prompt = args.get("prompt", "")
    if mode not in _VALID_MODES:
        raise ValueError(
            f"codex: invalid mode {mode!r} (must be one of {sorted(_VALID_MODES)})"
        )
    if not target:
        raise ValueError("codex: 'target' arg is required")

    stdout = invoke_codex(prompt, mode=mode, cwd=getattr(ctx, "sprint_dir", "."))

    # D-15: verdict only meaningful in 'review' mode; heuristic — look for "LGTM"/"pass"
    # at the start of stdout. For adversarial/consultation, verdict is n/a.
    verdict: Literal["pass", "fail", "n/a"]
    if mode != "review":
        verdict = "n/a"
    else:
        first_line = stdout.strip().splitlines()[0].lower() if stdout.strip() else ""
        if any(kw in first_line for kw in ("lgtm", "pass", "approved")):
            verdict = "pass"
        elif any(kw in first_line for kw in ("fail", "reject", "block")):
            verdict = "fail"
        else:
            verdict = "n/a"

    # Keep summary reasonable — first 500 chars of stdout
    summary = stdout.strip()[:500]

    created_at = datetime.now(timezone.utc).isoformat()
    schema = CodexReview(
        artifact_type="codex-review",
        mode=mode,
        target=target,
        verdict=verdict,
        summary=summary,
        sprint_id=getattr(ctx, "sprint_id", ""),
        created_at=created_at,
        persona=role,
    )

    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_dir.mkdir(parents=True, exist_ok=True)
    artifact_path = sprint_dir / "codex-review.md"
    atomic_write_text(artifact_path, _frontmatter_yaml(schema))

    return {
        "artifact_path": str(artifact_path),
        "mode": mode,
        "stdout": stdout,
    }
```
