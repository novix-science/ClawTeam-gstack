```python
"""Questionary-backed interactive wizard for /setup-deploy (SKILL-19).

Pure UI module — no filesystem side effects. Handler owns the write.
"""
from __future__ import annotations
import re
from typing import Any

_PROJECT_SLUG_RE = re.compile(r"^[a-zA-Z0-9_-]+$")
_SHELL_METACHARS = frozenset(";", "|", "&", "`", "$", "<", ">", "\\", "\n") if False else frozenset((";", "|", "&", "`", "$", "<", ">", "\\", "\n"))


def validate_project_slug(s: str | None) -> bool:
    """Accept only alphanumeric + _ and -; reject empty/None."""
    if not s:
        return False
    return bool(_PROJECT_SLUG_RE.fullmatch(s))


def validate_custom_cmd(s: str | None) -> bool:
    """Reject shell metachars; accept empty + simple commands + args."""
    if s is None:
        return True  # empty is fine
    return not any(c in s for c in _SHELL_METACHARS)


def _load_questionary() -> Any:
    """Lazy-import questionary (already in base deps; matches commands.py:173-183 pattern)."""
    try:
        import questionary  # noqa: F401
        return questionary
    except ImportError as exc:
        raise RuntimeError(
            "questionary is required but missing; pip install 'questionary>=2.0.1,<3.0.0'"
        ) from exc


def run_wizard() -> dict[str, Any]:
    """Run the /setup-deploy wizard; return a dict ready for DeployConfig(**result).

    Raises KeyboardInterrupt on user Ctrl-C (questionary returns None → re-raise).
    """
    questionary = _load_questionary()

    provider = questionary.select(
        "Which deploy provider?",
        choices=["vercel", "netlify", "fly", "custom"],
    ).ask()
    if provider is None:
        raise KeyboardInterrupt

    project = questionary.text(
        "Project slug (alphanumeric + dash/underscore only):",
        validate=lambda s: validate_project_slug(s) or "Invalid: ^[a-zA-Z0-9_-]+$",
    ).ask()
    if project is None:
        raise KeyboardInterrupt

    custom_cmd = ""
    if provider == "custom":
        custom_cmd = questionary.text(
            "Custom deploy command (no shell metachars):",
            validate=lambda s: validate_custom_cmd(s) or "No shell metachars: ; | & ` $ < > \\",
        ).ask()
        if custom_cmd is None:
            raise KeyboardInterrupt

    return {
        "provider": provider,
        "project": project,
        "custom_deploy_cmd": custom_cmd,
    }


def confirm_overwrite() -> bool:
    """Yes/no confirmation for idempotent re-run."""
    questionary = _load_questionary()
    answer = questionary.confirm(
        "Existing [deploy] block found. Overwrite?", default=False,
    ).ask()
    return bool(answer)
```

**3. `clawteam/templates/gstack/skills/setup_deploy/handler.py` (NEW, ~100 LOC):**

```python
"""/setup-deploy handler — writes [deploy] block into gstack.toml atomically."""
from __future__ import annotations
import re
import tomllib
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.templates import DeployConfig
from clawteam.templates.gstack.skills.setup_deploy.wizard import (
    run_wizard, confirm_overwrite,
)

# Match the entire [deploy]... block up to (but not including) the next [section] or EOF
_DEPLOY_BLOCK_RE = re.compile(
    r"(?ms)^\[deploy\]\s*\n(?:(?!^\[).*\n?)*",
)


def _find_gstack_toml(ctx: Any) -> Path:
    """Locate gstack.toml. Handler-local resolution — prefers ctx.gstack_toml_path."""
    path = getattr(ctx, "gstack_toml_path", None)
    if path is not None:
        return Path(path)
    # Fallback: conventional location inside workspace
    workspace = Path(getattr(ctx, "workspace_dir", "."))
    candidate = workspace / "gstack.toml"
    if candidate.exists():
        return candidate
    # Project-root alternate
    return workspace / "clawteam" / "templates" / "gstack.toml"


def _block_exists(content: str) -> bool:
    return bool(_DEPLOY_BLOCK_RE.search(content))


def _render_block(cfg: DeployConfig) -> str:
    """Render [deploy] block as TOML lines."""
    lines = [
        "[deploy]",
        f'provider = "{cfg.provider}"',
        f'project = "{cfg.project}"',
    ]
    if cfg.custom_deploy_cmd:
        lines.append(f'custom_deploy_cmd = "{cfg.custom_deploy_cmd}"')
    return "\n".join(lines) + "\n"


def _insert_or_replace_block(content: str, new_block: str) -> str:
    """Replace existing [deploy] block in place, or append to EOF with blank-line sep."""
    if _block_exists(content):
        return _DEPLOY_BLOCK_RE.sub(new_block, content, count=1)
    # Append with separating blank line if content doesn't end in \n\n
    sep = "\n" if content.endswith("\n") else "\n\n"
    return content + sep + new_block


def setup_deploy_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/setup-deploy entry point."""
    toml_path = _find_gstack_toml(ctx)
    if not toml_path.exists():
        raise FileNotFoundError(f"gstack.toml not found at {toml_path}")

    existing_content = toml_path.read_text()
    existing_block = _block_exists(existing_content)
    # args.skip_confirm flag for tests
    skip_confirm = bool(args.get("_skip_confirm", False))

    if existing_block and not skip_confirm:
        if not confirm_overwrite():
            return {
                "status": "skipped",
                "reason": "user declined overwrite",
                "gstack_toml": str(toml_path),
            }

    raw_cfg = run_wizard()
    # pydantic validation rejects invalid provider etc. — raises ValidationError.
    cfg = DeployConfig(**raw_cfg)
    new_block = _render_block(cfg)
    new_content = _insert_or_replace_block(existing_content, new_block)

    atomic_write_text(toml_path, new_content)

    # Verify the write parses back via tomllib
    try:
        parsed = tomllib.loads(new_content)
    except tomllib.TOMLDecodeError as exc:
        raise RuntimeError(f"setup-deploy produced invalid TOML: {exc}") from exc
    assert parsed.get("deploy", {}).get("provider") == cfg.provider

    return {
        "status": "written",
        "provider": cfg.provider,
        "project": cfg.project,
        "gstack_toml": str(toml_path),
    }
```
