```python
"""/benchmark handler (SKILL-18, D-10).

Lighthouse primary, curl-w fallback. Writes benchmark-report.md + optional
baseline JSON (consumed by /canary). Emits WebVitalRegressionDetected when
any vital exceeds threshold_ratio * baseline.
"""
from __future__ import annotations
import json
import shutil
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.events.types import WebVitalRegressionDetected
from clawteam.plugins.skill_errors import SkillPreconditionError
from clawteam.spawn.invoke import invoke_native_cli
from clawteam.templates.gstack.schemas.benchmark_report import BenchmarkReport
from clawteam.templates.gstack.skills.canary.poller import baseline_path, load_baseline

_DEFAULT_THRESHOLD_RATIO = 1.5
_VITAL_KEYS = ("lcp_ms", "fid_ms", "cls_score", "ttfb_ms", "dom_loaded_ms")


def _read_deploy_notes(sprint_dir: Path) -> dict[str, Any]:
    path = sprint_dir / "deploy.md"
    if not path.exists():
        raise SkillPreconditionError(
            skill="/benchmark", role="sre",
            message="Run /land-and-deploy first (deploy.md not found)",
        )
    text = path.read_text()
    if not text.startswith("---"):
        raise SkillPreconditionError(
            skill="/benchmark", role="sre", message="deploy.md has no frontmatter",
        )
    end = text.find("\n---", 3)
    if end == -1:
        raise SkillPreconditionError(
            skill="/benchmark", role="sre", message="deploy.md frontmatter unterminated",
        )
    fm: dict[str, Any] = {}
    for line in text[3:end].splitlines():
        if ":" not in line:
            continue
        k, _, v = line.partition(":")
        v = v.strip()
        if v.startswith(("'", '"')) and v.endswith(("'", '"')) and len(v) >= 2:
            v = v[1:-1]
        fm[k.strip()] = v
    return fm


def _lighthouse_available() -> bool:
    return shutil.which("lighthouse") is not None


def _run_lighthouse(url: str, timeout: int = 180) -> dict[str, float | None]:
    """Run lighthouse; return {lcp_ms, fid_ms, cls_score, ttfb_ms} — missing keys → None."""
    result = invoke_native_cli(
        ["lighthouse", url, "--output=json", "--quiet",
         "--chrome-flags=--headless"],
        timeout=timeout,
    )
    stdout = result.stdout.strip()
    if not stdout.startswith("{"):
        return {}
    try:
        data = json.loads(stdout)
    except json.JSONDecodeError:
        return {}
    audits = data.get("audits", {})
    def _audit(key: str) -> float | None:
        a = audits.get(key)
        if not isinstance(a, dict):
            return None
        v = a.get("numericValue")
        return float(v) if isinstance(v, (int, float)) else None
    return {
        "lcp_ms": _audit("largest-contentful-paint"),
        "fid_ms": _audit("max-potential-fid"),
        "cls_score": _audit("cumulative-layout-shift"),
        "ttfb_ms": _audit("server-response-time"),
    }


def _run_curl(url: str, timeout: int = 30) -> dict[str, float | None]:
    """Curl fallback: returns {ttfb_ms, dom_loaded_ms}; lcp/fid/cls → None."""
    # -w '%{time_total} %{time_starttransfer}' → two floats in seconds
    result = invoke_native_cli(
        ["curl", "-o", "/dev/null", "-s", "-w", "%{time_total} %{time_starttransfer}", url],
        timeout=timeout,
    )
    parts = result.stdout.strip().split()
    try:
        time_total = float(parts[0]) * 1000  # → ms
        time_starttransfer = float(parts[1]) * 1000
    except (IndexError, ValueError):
        return {}
    return {
        "ttfb_ms": time_starttransfer,
        "dom_loaded_ms": time_total,
    }


def _collect_vitals(url: str) -> tuple[dict[str, float | None], str]:
    """Return (vitals_dict, measured_with).

    Try lighthouse first. On missing binary OR parse failure, fall back to curl.
    """
    if _lighthouse_available():
        vitals = _run_lighthouse(url)
        if vitals:
            # Fill missing ttfb_ms + dom_loaded_ms via curl if lighthouse omitted
            if not vitals.get("ttfb_ms") or not vitals.get("dom_loaded_ms"):
                curl = _run_curl(url)
                if curl.get("ttfb_ms") is not None and vitals.get("ttfb_ms") is None:
                    vitals["ttfb_ms"] = curl["ttfb_ms"]
                if curl.get("dom_loaded_ms") is not None:
                    vitals["dom_loaded_ms"] = curl["dom_loaded_ms"]
            return vitals, "lighthouse"
    # Fallback
    vitals = _run_curl(url)
    return vitals, "curl"


def _evaluate_regressions(
    current: dict[str, float | None],
    baseline: dict[str, float] | None,
    threshold_ratio: float,
) -> list[tuple[str, float, float, float]]:
    """Return list of (vital, baseline_value, observed_value, ratio) for regressions."""
    if not baseline:
        return []
    hits: list[tuple[str, float, float, float]] = []
    for vital in ("lcp_ms", "fid_ms", "cls_score", "ttfb_ms", "dom_loaded_ms"):
        observed = current.get(vital)
        base = baseline.get(vital)
        if observed is None or base is None or base <= 0:
            continue
        ratio = observed / base
        if ratio > threshold_ratio:
            hits.append((vital.replace("_ms", "").replace("_score", ""), base, observed, ratio))
    return hits


def _write_report(
    sprint_dir: Path,
    *,
    sprint_id: str,
    vitals: dict[str, float | None],
    measured_with: str,
    benchmark_status: str,
    regression_flags: list[str],
) -> Path:
    schema = BenchmarkReport(
        artifact_type="benchmark-report",
        benchmark_status=benchmark_status,
        lcp_ms=vitals.get("lcp_ms"),
        fid_ms=vitals.get("fid_ms"),
        cls_score=vitals.get("cls_score"),
        ttfb_ms=vitals.get("ttfb_ms") or 0.0,
        dom_loaded_ms=vitals.get("dom_loaded_ms") or 0.0,
        regression_flags=regression_flags,
        measured_with=measured_with if measured_with in ("lighthouse", "curl") else "curl",
        sprint_id=sprint_id,
        created_at=datetime.now(timezone.utc).isoformat(),
    )
    d = schema.model_dump()
    lines = ["---"]
    for k, v in d.items():
        if isinstance(v, bool):
            lines.append(f"{k}: {str(v).lower()}")
        elif isinstance(v, (list, dict)):
            lines.append(f"{k}: {json.dumps(v)}")
        elif isinstance(v, str):
            lines.append(f"{k}: {v!r}")
        elif v is None:
            lines.append(f"{k}: null")
        else:
            lines.append(f"{k}: {v}")
    lines.append("---")
    lines.append("")
    artifact_path = sprint_dir / "benchmark-report.md"
    sprint_dir.mkdir(parents=True, exist_ok=True)
    atomic_write_text(artifact_path, "\n".join(lines) + "\n")
    return artifact_path


def _write_baseline(
    team: str, provider: str, vitals: dict[str, float | None],
) -> Path:
    """Write pre-deploy baseline JSON consumed by /canary."""
    path = baseline_path(team, provider)
    path.parent.mkdir(parents=True, exist_ok=True)
    # Use avg_response_ms proxy = ttfb + dom_loaded_ms / 2 if present
    ttfb = vitals.get("ttfb_ms") or 0.0
    dom = vitals.get("dom_loaded_ms") or 0.0
    avg = (ttfb + dom) / 2 if (ttfb or dom) else 0.0
    data = {k: vitals.get(k) for k in _VITAL_KEYS}
    data["avg_response_ms"] = avg
    atomic_write_text(path, json.dumps(data, indent=2))
    return path


def benchmark_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/benchmark entry point (sre role)."""
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    team_name = args.get("team") or getattr(ctx, "team_name", "default")

    deploy_fm = _read_deploy_notes(sprint_dir)
    deploy_url = deploy_fm.get("deploy_url", "")
    provider = deploy_fm.get("provider", "custom")

    tmpl = getattr(ctx, "template", None) or getattr(ctx, "tmpl", None)
    bench_cfg = getattr(tmpl, "benchmark", None) if tmpl is not None else None
    threshold_ratio = (
        float(bench_cfg.regression_threshold_ratio)
        if bench_cfg is not None else _DEFAULT_THRESHOLD_RATIO
    )

    vitals, measured_with = _collect_vitals(deploy_url)

    # Optionally write baseline
    is_pre_deploy = bool(args.get("pre_deploy", False))
    if is_pre_deploy:
        _write_baseline(team_name, provider, vitals)

    # Regression eval against existing baseline
    baseline = load_baseline(team_name, provider) if not is_pre_deploy else None
    hits = _evaluate_regressions(vitals, baseline, threshold_ratio)
    regression_flags = [f"{vital}>{threshold_ratio}x_baseline" for vital, _, _, _ in hits]
    benchmark_status = "regression" if regression_flags else "clean"

    # Emit event(s) per regression
    bus = getattr(ctx, "bus", None)
    if bus is not None:
        for vital_name, base_val, obs_val, ratio in hits:
            bus.emit(WebVitalRegressionDetected(
                team_name=team_name, sprint_id=sprint_id,
                deploy_url=deploy_url, vital=vital_name,
                baseline_value=base_val, observed_value=obs_val, ratio=ratio,
            ))

    artifact = _write_report(
        sprint_dir, sprint_id=sprint_id, vitals=vitals,
        measured_with=measured_with, benchmark_status=benchmark_status,
        regression_flags=regression_flags,
    )
    return {
        "artifact_path": str(artifact),
        "benchmark_status": benchmark_status,
        "measured_with": measured_with,
        "regression_flags": regression_flags,
    }
```
