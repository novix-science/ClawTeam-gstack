```python
"""Framework detection from package.json (D-13)."""
from __future__ import annotations

import json
from dataclasses import dataclass, field
from pathlib import Path

_FRAMEWORK_KEYS = ["react", "svelte", "vue"]  # detection order per D-13


@dataclass
class FrameworkDetection:
    framework: str = ""  # "react" | "svelte" | "vue" | "plain" | "" (ambiguous)
    source_root: str = ""
    ambiguous: bool = False
    candidates: list[str] = field(default_factory=list)


def _read_package_json(project_root: Path) -> dict | None:
    pkg = project_root / "package.json"
    if not pkg.is_file():
        return None
    try:
        return json.loads(pkg.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, OSError):
        return None


def _collect_all_deps(pkg: dict) -> set[str]:
    all_deps = set()
    all_deps.update((pkg.get("dependencies") or {}).keys())
    all_deps.update((pkg.get("devDependencies") or {}).keys())
    return all_deps


def detect_framework(project_root: Path) -> FrameworkDetection:
    """Detect the frontend framework by inspecting package.json deps.

    Per D-13:
      1. No package.json -> plain HTML.
      2. Exactly one of {react, svelte, vue} present -> that framework.
      3. Multiple present -> ambiguous; caller writes question.md.
      4. None present -> plain HTML.
    """
    pkg = _read_package_json(project_root)
    if pkg is None:
        return FrameworkDetection(framework="plain", source_root="", ambiguous=False)

    deps = _collect_all_deps(pkg)
    present = [k for k in _FRAMEWORK_KEYS if k in deps]

    if len(present) == 0:
        return FrameworkDetection(framework="plain", source_root="", ambiguous=False)
    if len(present) == 1:
        return FrameworkDetection(
            framework=present[0], source_root="src", ambiguous=False,
        )
    return FrameworkDetection(
        framework="", source_root="", ambiguous=True, candidates=present,
    )


__all__ = ["FrameworkDetection", "detect_framework"]
```

```python
"""/design-html handler (SKILL-12, D-13).

Reads a mockup HTML file, detects the project framework, emits production
source at the framework's conventional source root.
"""
from __future__ import annotations

import hashlib
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text
from clawteam.templates.gstack.skills.design_html.framework_detect import (
    FrameworkDetection, detect_framework,
)


def _emit_react(src_dir: Path, component_name: str, mockup_html: str) -> Path:
    target = src_dir / f"{component_name}.jsx"
    content = (
        f"// Auto-emitted by /design-html ({component_name}.jsx)\n"
        f"// Source mockup hash: {hashlib.sha256(mockup_html.encode()).hexdigest()[:8]}\n"
        "import React from 'react';\n\n"
        f"export default function {component_name}() {{\n"
        "  return (\n"
        "    <div dangerouslySetInnerHTML={{ __html: MOCKUP_HTML }} />\n"
        "  );\n"
        "}\n\n"
        f"const MOCKUP_HTML = {mockup_html!r};\n"
    )
    target.parent.mkdir(parents=True, exist_ok=True)
    atomic_write_text(target, content)
    return target


def _emit_svelte(src_dir: Path, component_name: str, mockup_html: str) -> Path:
    target = src_dir / f"{component_name}.svelte"
    content = (
        f"<!-- Auto-emitted by /design-html ({component_name}.svelte) -->\n"
        f"<script>\n  // Mockup hash: {hashlib.sha256(mockup_html.encode()).hexdigest()[:8]}\n</script>\n\n"
        f"{mockup_html}\n"
    )
    target.parent.mkdir(parents=True, exist_ok=True)
    atomic_write_text(target, content)
    return target


def _emit_vue(src_dir: Path, component_name: str, mockup_html: str) -> Path:
    target = src_dir / f"{component_name}.vue"
    content = (
        f"<!-- Auto-emitted by /design-html ({component_name}.vue) -->\n"
        f"<template>\n{mockup_html}\n</template>\n\n"
        f"<script>\nexport default {{ name: '{component_name}' }};\n</script>\n"
    )
    target.parent.mkdir(parents=True, exist_ok=True)
    atomic_write_text(target, content)
    return target


def _emit_plain(project_root: Path, mockup_html: str) -> list[Path]:
    paths: list[Path] = []
    # index.html at project root
    index = project_root / "index.html"
    atomic_write_text(index, mockup_html)
    paths.append(index)
    # styles.css (empty scaffolding)
    styles = project_root / "styles.css"
    atomic_write_text(styles, "/* Auto-emitted by /design-html */\n")
    paths.append(styles)
    # app.js (empty scaffolding)
    app = project_root / "app.js"
    atomic_write_text(app, "// Auto-emitted by /design-html\n")
    paths.append(app)
    return paths


def _write_ambiguity_question(
    sprint_dir: Path, candidates: list[str], mockup_hash: str,
) -> Path:
    questions_dir = sprint_dir / "questions"
    questions_dir.mkdir(parents=True, exist_ok=True)
    qid = f"design-html-framework-{mockup_hash[:8]}"
    body = (
        f"---\nquestion_id: {qid}\npriority: blocking\n---\n\n"
        "### Multiple frontend frameworks detected\n\n"
        f"`package.json` references multiple frontend frameworks: "
        f"{', '.join(sorted(candidates))}.\n\n"
        "Which target should /design-html emit for?\n\n"
        + "\n".join(f"- {c}" for c in sorted(candidates))
        + "\n\nReply in `answers/" + qid + ".md` with a `choice: <framework>` field.\n"
    )
    q_path = questions_dir / f"{qid}.md"
    atomic_write_text(q_path, body)
    return q_path


def _write_note(
    sprint_dir: Path,
    *,
    status: str,
    framework: str,
    target_paths: list[str],
    role: str,
    sprint_id: str,
) -> Path:
    created_at = datetime.now(timezone.utc).isoformat()
    lines = [
        "---",
        "artifact_type: design-html-note",
        f"status: {status!r}",
        f"framework: {framework!r}",
        f"target_paths: {target_paths!r}",
        f"sprint_id: {sprint_id!r}",
        f"created_at: {created_at!r}",
        f"persona: {role!r}",
        "step_label: design-html",
        "done: true",
        "---",
        "",
    ]
    note = sprint_dir / "design-html-note.md"
    atomic_write_text(note, "\n".join(lines) + "\n")
    return note


def design_html_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/design-html entry point."""
    mockup_path = Path(args["mockup_html_path"])
    if not mockup_path.is_file():
        raise FileNotFoundError(f"mockup not found: {mockup_path}")
    mockup_html = mockup_path.read_text(encoding="utf-8")

    project_root = Path(
        args.get("project_root") or getattr(ctx, "project_root", ".")
    )
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    component_name = args.get("component_name", "Hero")

    detection = detect_framework(project_root)

    if detection.ambiguous:
        mockup_hash = hashlib.sha256(mockup_html.encode()).hexdigest()
        q = _write_ambiguity_question(sprint_dir, detection.candidates, mockup_hash)
        note = _write_note(
            sprint_dir, status="awaiting_user", framework="",
            target_paths=[str(q)], role=role, sprint_id=sprint_id,
        )
        return {
            "status": "awaiting_user",
            "candidates": detection.candidates,
            "question_path": str(q),
            "artifact_path": str(note),
        }

    fw = detection.framework
    src_dir = project_root / detection.source_root if detection.source_root else project_root
    targets: list[Path] = []
    if fw == "react":
        targets.append(_emit_react(src_dir, component_name, mockup_html))
    elif fw == "svelte":
        targets.append(_emit_svelte(src_dir, component_name, mockup_html))
    elif fw == "vue":
        targets.append(_emit_vue(src_dir, component_name, mockup_html))
    else:
        targets.extend(_emit_plain(project_root, mockup_html))

    target_paths = [str(t) for t in targets]
    note = _write_note(
        sprint_dir, status="emitted", framework=fw,
        target_paths=target_paths, role=role, sprint_id=sprint_id,
    )
    return {
        "status": "emitted",
        "framework": fw,
        "target_paths": target_paths,
        "artifact_path": str(note),
    }


__all__ = ["design_html_handler"]
```
