```python
from __future__ import annotations
from pathlib import Path
from clawteam.memory.entry import MemoryEntry
from clawteam.utils.io import atomic_write_text, file_locked

HIGH_IMPACT_TAG = "impact:high"
DECISIONS_SCOPE_PREFIX = "team/decisions"

def check_high_impact(entry: MemoryEntry) -> tuple[bool, str]:
    """Return (requires_confirmation, reason)."""
    if HIGH_IMPACT_TAG in entry.tags:
        return True, f"tagged {HIGH_IMPACT_TAG}"
    if entry.scope == "team" and any(t.startswith("decisions") for t in entry.tags):
        return True, "memory/team/decisions/ scope"
    return False, ""

def stage_pending(entry: MemoryEntry, team: str, sprint_id: str | None, root: Path) -> Path:
    """Stage entry to memory/<scope>/pending/<id>.jsonl + write memory-confirm-<id>.md question artifact."""
    pending_dir = root / "teams" / team / "memory" / entry.scope / "pending"
    pending_dir.mkdir(parents=True, exist_ok=True)
    pending_path = pending_dir / f"{entry.id}.jsonl"
    with file_locked(pending_path.with_suffix(".lock")):
        atomic_write_text(pending_path, entry.model_dump_json() + "\n")
    if sprint_id:
        questions_dir = root / "teams" / team / "sprints" / sprint_id / "questions"
        questions_dir.mkdir(parents=True, exist_ok=True)
        question_path = questions_dir / f"memory-confirm-{entry.id}.md"
        body = (
            f"# Memory write: high-impact confirmation required\n\n"
            f"**Entry id:** {entry.id}\n"
            f"**Scope:** {entry.scope}\n"
            f"**Tags:** {', '.join(entry.tags)}\n"
            f"**Author:** {entry.author}\n"
            f"**Evidence:** {entry.evidence or '(none)'}\n\n"
            f"## Preview\n\n> {entry.body[:300]}...\n\n"
            f"## Options\n\n1. Confirm\n2. Edit\n3. Reject\n"
        )
        atomic_write_text(question_path, body)
    return pending_path
```

```python
from __future__ import annotations
import math
import re
from dataclasses import dataclass
from clawteam.memory.entry import MemoryEntry
from clawteam.memory.store import TeamMemoryStore

_TOKEN_RE = re.compile(r"[A-Za-z0-9_]+")

def _bag_of_words(text: str) -> dict[str, int]:
    bow: dict[str, int] = {}
    for tok in _TOKEN_RE.findall(text.lower()):
        bow[tok] = bow.get(tok, 0) + 1
    return bow

def _cosine(a: dict[str, int], b: dict[str, int]) -> float:
    keys = set(a) | set(b)
    dot = sum(a.get(k, 0) * b.get(k, 0) for k in keys)
    na = math.sqrt(sum(v * v for v in a.values()))
    nb = math.sqrt(sum(v * v for v in b.values()))
    if na == 0 or nb == 0:
        return 0.0
    return dot / (na * nb)

@dataclass(frozen=True)
class ConflictMatch:
    other_id: str
    similarity: float
    shared_tags: tuple[str, ...]

def detect_conflicts(
    new_entry: MemoryEntry,
    store: TeamMemoryStore,
    *,
    threshold: float = 0.75,
) -> list[ConflictMatch]:
    """Return existing entries sharing any tag with new_entry where cosine(bow) >= threshold."""
    if not new_entry.tags:
        return []
    new_bow = _bag_of_words(new_entry.body)
    # list() exposes all entries in same scope (role-scoped entries stay per-role)
    matches: list[ConflictMatch] = []
    for existing in store.list(scope=new_entry.scope, role=new_entry.role):
        if existing.id == new_entry.id:
            continue
        shared = set(new_entry.tags) & set(existing.tags)
        if not shared:
            continue
        sim = _cosine(new_bow, _bag_of_words(existing.body))
        if sim >= threshold:
            matches.append(ConflictMatch(existing.id, sim, tuple(sorted(shared))))
    return matches
```

```python
from __future__ import annotations
import json
from pathlib import Path
from clawteam.memory.entry import MemoryEntry
from clawteam.memory.store import TeamMemoryStore

def backfill_scan(team: str, root: Path) -> int:
    """Walk _phase6_pending/*.json, promote to memory/team/retros/.
    Idempotent: checks .processed sentinel. Returns count of promoted entries."""
    pending_dir = root / "teams" / team / "_phase6_pending"
    if not pending_dir.exists():
        return 0
    processed_dir = pending_dir / ".processed"
    processed_dir.mkdir(exist_ok=True)
    store = TeamMemoryStore(team=team, data_dir=root)
    count = 0
    for retro_path in sorted(pending_dir.glob("*.json")):
        sentinel = processed_dir / (retro_path.stem + ".processed")
        if sentinel.exists():
            continue
        try:
            data = json.loads(retro_path.read_text())
        except Exception:
            continue
        entry = MemoryEntry(
            id=f"retro-{retro_path.stem}",
            author=data.get("author", f"reflect@{retro_path.stem}"),
            sprint_id=data.get("sprint_id", retro_path.stem),
            phase="reflect",
            timestamp=data.get("timestamp"),
            tags=["retro", "phase-reflect"],
            evidence=data.get("evidence", f"_phase6_pending/{retro_path.name}"),
            confidence=0.8,
            learned_from="artifact",
            scope="team",
            role=None,
            body=data.get("body", "") or json.dumps(data)[:2000],
            op="write",
        )
        try:
            store.write(entry)
        except Exception:
            continue
        sentinel.write_text("processed")
        count += 1
    return count
```

Hook into `learn_handler`: first invocation per `team` triggers `backfill_scan(team, root)` once per process. Use a simple per-process `set[str]` sentinel `_backfilled_teams` keyed by team name.
