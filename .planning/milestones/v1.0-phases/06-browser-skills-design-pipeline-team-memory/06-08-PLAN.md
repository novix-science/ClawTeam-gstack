Key structure:

```python
"""/design-shotgun handler (SKILL-11, D-12).

Multi-turn state machine orchestration. Each invocation carries an
`action` arg (init|generate|publish|pick|refine|converge|abandon) that
maps to DSEvent for the state transition. State persists at
<sprint_dir>/design-shotgun-state.json under file_locked.
"""
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

from clawteam.fileutil import atomic_write_text, file_locked
from clawteam.memory import MemoryEntry, TeamMemoryStore
from clawteam.templates.gstack.skills.design_shotgun.state import (
    DSEvent, DSState, ShotgunState, VariantFixture,
)

_DEFAULT_VARIANT_COUNT = 4


def _resolve_variant_count(ctx: Any) -> int:
    tmpl = getattr(ctx, "template", None) or getattr(ctx, "tmpl", None)
    if tmpl is not None:
        cfg = getattr(tmpl, "design_shotgun", None)
        if cfg is not None:
            return int(cfg.variant_count)
    return _DEFAULT_VARIANT_COUNT


def _state_path(sprint_dir: Path) -> Path:
    return sprint_dir / "design-shotgun-state.json"


def _load_state(sprint_dir: Path) -> ShotgunState:
    path = _state_path(sprint_dir)
    if not path.is_file():
        return ShotgunState()
    try:
        raw = json.loads(path.read_text(encoding="utf-8"))
        return ShotgunState.from_json_dict(raw)
    except (json.JSONDecodeError, OSError, ValueError):
        return ShotgunState()


def _save_state(sprint_dir: Path, state: ShotgunState) -> None:
    path = _state_path(sprint_dir)
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = json.dumps(state.to_json_dict(), indent=2)
    with file_locked(path):
        atomic_write_text(path, payload)


def _write_board(sprint_dir: Path, variants: list[VariantFixture]) -> Path:
    board_dir = sprint_dir / "design-board"
    board_dir.mkdir(parents=True, exist_ok=True)
    lines = ["# Design Shotgun Board", ""]
    for v in variants:
        variant_dir = board_dir / v.variant_id
        variant_dir.mkdir(parents=True, exist_ok=True)
        # HTML content is authored by the designer-role LLM and passed in;
        # handler only writes it. (Production: generate via designer's turn;
        # tests: load from tests/fixtures/design_shotgun/.)
        html_target = variant_dir / "index.html"
        if v.html_path:
            source = Path(v.html_path)
            if source.is_file():
                html_target.write_text(source.read_text(encoding="utf-8"))
            else:
                html_target.write_text(f"<!-- no source: {v.html_path} -->")
        lines.append(f"## {v.title}")
        lines.append(f"- variant_id: {v.variant_id}")
        lines.append(f"- HTML: design-board/{v.variant_id}/index.html")
        lines.append(f"- {v.description}")
        lines.append("")
    index = board_dir / "index.md"
    atomic_write_text(index, "\n".join(lines) + "\n")
    return index


def _write_taste_memory(
    team_name: str,
    *,
    sprint_id: str,
    picked_variant_id: str,
    reason: str,
) -> str:
    store = TeamMemoryStore(team_name)
    entry_id = store.gen_id(
        author="designer@design-shotgun",
        title=f"picked {picked_variant_id}",
        tags=["design", "taste", "design-shotgun"],
    )
    entry = MemoryEntry(
        id=entry_id,
        author="designer@design-shotgun",
        sprint_id=sprint_id,
        phase="build",
        title=f"Taste observation: picked {picked_variant_id}",
        body=(
            f"User picked {picked_variant_id} from a {reason!r}: {reason}"
        ),
        tags=["design", "taste", "design-shotgun"],
        evidence=f"design-board/{picked_variant_id}/index.html",
        confidence=0.9,
        learned_from="user",  # user picked; high-provenance
        scope="role",
        role="designer",
    )
    store.write(entry)
    return entry_id


def _write_note(sprint_dir: Path, *, state: ShotgunState, status: str) -> Path:
    created_at = datetime.now(timezone.utc).isoformat()
    lines = [
        "---",
        "artifact_type: design-shotgun-note",
        f"status: {status!r}",
        f"state: {state.state.value!r}",
        f"iteration: {state.iteration}",
        f"variant_count: {state.variant_count}",
        f"picked_variant_id: {state.picked_variant_id!r}",
        f"sprint_id: {state.sprint_id!r}",
        f"created_at: {created_at!r}",
        "persona: designer",
        "step_label: design-shotgun",
        "done: true",
        "---",
        "",
    ]
    artifact = sprint_dir / "design-shotgun-note.md"
    atomic_write_text(artifact, "\n".join(lines) + "\n")
    return artifact


_ACTION_TO_EVENT = {
    "init": None,  # handled specially (no transition; initializes state)
    "generate": DSEvent.GENERATE_REQUESTED,
    "variants_ready": DSEvent.VARIANTS_READY,
    "publish": DSEvent.BOARD_PUBLISHED,
    "pick": DSEvent.USER_PICKED,
    "refine": DSEvent.REFINE_REQUESTED,
    "converge": DSEvent.CONVERGE,
    "abandon": DSEvent.ABANDON,
}


def shotgun_handler(ctx: Any, *, role: str, args: dict[str, Any]) -> dict[str, Any]:
    """/design-shotgun entry point — per-turn state transition."""
    sprint_dir = Path(getattr(ctx, "sprint_dir", "."))
    sprint_id = getattr(ctx, "sprint_id", "")
    team_name = getattr(ctx, "team_name", "default")

    state = _load_state(sprint_dir)
    state.sprint_id = state.sprint_id or sprint_id

    action = args.get("action", "init")
    if action == "init":
        state.variant_count = _resolve_variant_count(ctx)
        _save_state(sprint_dir, state)
        return {
            "status": "initialized",
            "state": state.state.value,
            "variant_count": state.variant_count,
        }

    event = _ACTION_TO_EVENT.get(action)
    if event is None:
        raise ValueError(f"unknown action {action!r}")

    # Collect transition kwargs
    kwargs: dict[str, Any] = {}
    if action == "variants_ready":
        raw_variants = args.get("variants") or []
        kwargs["variants"] = [
            v if isinstance(v, VariantFixture) else VariantFixture(**v)
            for v in raw_variants
        ]
    elif action == "pick":
        kwargs["picked_variant_id"] = args["variant_id"]
        kwargs["pick_reason"] = args.get("reason", "")

    new_state = state.advance(event, **kwargs)

    # Side effects per new state
    memory_id = ""
    if action == "generate":
        # Optionally the caller passes pre-rendered variants (test path) in
        # the same action; else defer to a subsequent variants_ready action.
        pre_rendered = args.get("variants")
        if pre_rendered:
            new_state = new_state.advance(DSEvent.VARIANTS_READY,
                                          variants=[VariantFixture(**v) for v in pre_rendered])
            _write_board(sprint_dir, new_state.variants)
    elif action == "variants_ready":
        _write_board(sprint_dir, new_state.variants)
    elif action == "pick":
        memory_id = _write_taste_memory(
            team_name,
            sprint_id=sprint_id,
            picked_variant_id=new_state.picked_variant_id,
            reason=new_state.pick_reason,
        )

    _save_state(sprint_dir, new_state)
    note = _write_note(sprint_dir, state=new_state, status=action)
    return {
        "status": action,
        "state": new_state.state.value,
        "iteration": new_state.iteration,
        "variant_count": new_state.variant_count,
        "picked_variant_id": new_state.picked_variant_id,
        "memory_id": memory_id,
        "artifact_path": str(note),
    }


__all__ = ["shotgun_handler"]
```
