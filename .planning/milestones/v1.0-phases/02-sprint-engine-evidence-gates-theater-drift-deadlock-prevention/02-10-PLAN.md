```python
import re
from typing import TYPE_CHECKING
if TYPE_CHECKING:
    from clawteam.events.bus import EventBus

# D-13 /careful destructive-command regex blacklist — warn-only by default
CAREFUL_BLACKLIST = re.compile(
    r"(?i)(rm\s+-[rR]f|git\s+reset\s+--hard|git\s+push\s+--force|DROP\s+TABLE|DELETE\s+FROM\s+.*\s+WHERE)"
)

# Module-level flag flipped by SprintConductor when careful_enabled=True on SprintState
_careful_veto_mode: bool = False

def set_careful_veto_mode(enabled: bool) -> None:
    """Set by SprintConductor on sprint start; read by the /careful subscriber."""
    global _careful_veto_mode
    _careful_veto_mode = enabled

def _on_before_file_write(event) -> None:
    """Subscriber: vetoes write when FreezeRegistry.is_frozen matches."""
    reg = get_freeze_registry()
    if reg is None:
        return
    frozen, reason = reg.is_frozen(event.path)
    if frozen:
        event.veto = True
        event.veto_reason = reason or f"{event.path} is /freeze-locked"

def _on_before_tool_call(event) -> None:
    """Subscriber: vetoes tool call when args contain a frozen path."""
    reg = get_freeze_registry()
    if reg is None:
        return
    # Shallow scan of string-valued args — deep recursion not needed per CONTEXT
    for k, v in (event.args or {}).items():
        if isinstance(v, str) and v.startswith("/"):
            frozen, reason = reg.is_frozen(v)
            if frozen:
                event.veto = True
                event.veto_reason = f"tool {event.tool_name}: {reason or 'path frozen'}"
                return

def _on_careful(event) -> None:
    """Subscriber: /careful blacklist regex on tool-call args. Warn-only unless veto mode."""
    # Combine all str args into one searchable string
    combined = " ".join(str(v) for v in (event.args or {}).values() if isinstance(v, str))
    m = CAREFUL_BLACKLIST.search(combined)
    if not m:
        return
    # Emit warn event regardless
    from clawteam.events.global_bus import get_event_bus
    from clawteam.events.types import FreezeChange
    get_event_bus().emit(FreezeChange(
        action="careful-warn",
        path="",
        agent=event.agent_name,
        reason=f"destructive pattern: {m.group(0)}",
        actor="careful-subscriber",
    ))
    if _careful_veto_mode:
        event.veto = True
        event.veto_reason = f"blocked by /careful: pattern {m.group(0)}"

_subscribers_registered: set[int] = set()

def register_safety_subscribers(bus) -> None:
    """Idempotent: register Before* subscribers for the given EventBus.

    Called by SprintConductor.__init__ — only for gstack sprints. Existing
    templates never call this so their EventBus has zero safety subscribers
    (Pitfall #8 BC hinge).
    """
    if id(bus) in _subscribers_registered:
        return
    from clawteam.events.types import BeforeFileWrite, BeforeToolCall
    bus.subscribe(BeforeFileWrite, _on_before_file_write, priority=10)
    bus.subscribe(BeforeToolCall, _on_before_tool_call, priority=10)
    bus.subscribe(BeforeToolCall, _on_careful, priority=20)
    _subscribers_registered.add(id(bus))

def reset_safety_subscribers() -> None:
    """Test helper — call in setUp/tearDown."""
    global _subscribers_registered, _careful_veto_mode
    _subscribers_registered = set()
    _careful_veto_mode = False
```

```python
def _tool(fn):
    @wraps(fn)
    def wrapped(*args, **kwargs):
        from clawteam.events.global_bus import get_event_bus
        from clawteam.events.types import BeforeToolCall
        from clawteam.identity import AgentIdentity
        identity = AgentIdentity.from_env()
        event = BeforeToolCall(
            team_name=identity.team_name or "",
            agent_name=identity.agent_name or "",
            tool_name=fn.__name__,
            args=dict(kwargs),
        )
        get_event_bus().emit(event)
        if event.veto:
            from clawteam.harness.freeze_registry import FrozenPathError
            raise FrozenPathError(event.veto_reason or f"tool {fn.__name__} vetoed by safety rail")
        try:
            return fn(*args, **kwargs)
        except Exception as exc:
            raise translate_error(exc) from exc
    wrapped.__signature__ = inspect.signature(fn)
    return mcp.tool()(wrapped)
```

def translate_error(exc: Exception) -> MCPToolError:
    if isinstance(exc, MCPToolError):
        return exc
    if isinstance(exc, TaskLockError):
        return MCPToolError(str(exc))
    # Phase 2 explicit branches (Plan 02-10, D-12)
    try:
        from clawteam.harness.freeze_registry import FrozenPathError
    except ImportError:
        FrozenPathError = None
    try:
        from clawteam.team.envelope import MalformedEnvelopeError
    except ImportError:
        MalformedEnvelopeError = None
    try:
        from clawteam.harness.artifacts import ArtifactTooLargeError
    except ImportError:
        ArtifactTooLargeError = None
    try:
        from clawteam.sprint.conductor import AmbiguousSprintError, SprintNotFoundError, MissingTeamError
    except ImportError:
        AmbiguousSprintError = SprintNotFoundError = MissingTeamError = None
    for cls in (FrozenPathError, MalformedEnvelopeError, ArtifactTooLargeError,
                AmbiguousSprintError, SprintNotFoundError, MissingTeamError):

        return MCPToolError(str(exc))
