<objective>
Ship the `SprintConductor` — the sprint-level loop owner that composes Phase 1's
`SprintState` + `PhaseRunner` primitives with Phase 2's new gates (`EvidenceGate`,
`forced_progress_gate`, `InteractionGate`) into a runnable sprint engine.

`SprintConductor` sits above `PhaseRunner`:
- `PhaseRunner._gates[phase]` holds the registered gates.
- `SprintConductor.advance_phase(sprint_id)` invokes `PhaseRunner.can_advance()` with
  the three-gate chain and writes the outcome.
- `SprintConductor.pause(sprint_id)` / `resume(sprint_id)` handle durable-state
  checkpointing + event replay (CORE-07).

Pause/resume is a strict CORE-07 requirement: the checkpoint survives a full
`HarnessOrchestrator` restart. We verify this via a subprocess test that:
1. Starts a sprint, advances past the first gate.
2. Pauses the sprint in the main process.
3. Spawns a fresh python subprocess that imports `SprintConductor` and calls
   `resume(sprint_id)` — confirms state.json + freeze.json + turn_counters all rehydrate.

INT-06 auto_advance semantics (D-23):
- `auto_advance=True` → `InteractionGate` only when unanswered questions exist.
- `auto_advance=False` → `InteractionGate` on every transition.
- `force_interactive_phases=["ship"]` (Phase 4) → always `InteractionGate` regardless.

Three-layer cap override (D-29):
- CLI flag > `CLAWTEAM_ARTIFACT_CAP_KB` env > `SprintState.artifact_cap_bytes` default.

```python
from __future__ import annotations
import uuid
from dataclasses import dataclass, field
from pathlib import Path
from threading import RLock
from typing import TYPE_CHECKING

from clawteam.fileutil import file_locked, atomic_write_text
from clawteam.team.models import get_data_dir
from clawteam.sprint.state import SprintState, save_sprint_state, load_sprint_state
from clawteam.identity import _env

if TYPE_CHECKING:
    from clawteam.events.bus import EventBus


class AmbiguousSprintError(ValueError):
    def __init__(self, prefix: str, candidates: list[str]):
        super().__init__(f"Ambiguous sprint prefix '{prefix}'; candidates: {candidates}")
        self.prefix = prefix
        self.candidates = candidates


class SprintNotFoundError(ValueError):
    def __init__(self, identifier: str):
        super().__init__(f"Sprint not found: {identifier}")
        self.identifier = identifier


class MissingTeamError(ValueError):
    def __init__(self, message: str = "Team name is required (--team or CLAWTEAM_TEAM env)"):
        super().__init__(message)


DEFAULT_FIRST_PHASE = "think"


class SprintConductor:
    """Per-team sprint-level loop owner. Composes PhaseRunner + SprintState.

    Phase 2 is single-sprint correctness; 10-concurrent is verified in Phase 7.
    """

    def __init__(
        self,
        team_name: str,
        *,
        force_interactive_phases: list[str] | None = None,
        artifact_cap_bytes: int | None = None,
        phase_artifact_cap_bytes: int | None = None,
        bus: "EventBus | None" = None,
    ):
        if not team_name:
            raise MissingTeamError()
        self.team_name = team_name
        self.force_interactive_phases = list(force_interactive_phases or [])
        self._lock = RLock()
        # Bus — default to global; tests inject their own
        if bus is None:
            from clawteam.events.global_bus import get_event_bus
            bus = get_event_bus()
        self.bus = bus
        # Activate safety rails (Plan 02-10) — opt-in via SprintConductor only
        from clawteam.harness.freeze_registry import register_safety_subscribers
        register_safety_subscribers(self.bus)
        # Resolve three-layer cap override (D-29)
        self._artifact_cap_override = self._resolve_cap(
            cli_override=artifact_cap_bytes,
            env_keys=("CLAWTEAM_ARTIFACT_CAP_KB", "OH_ARTIFACT_CAP_KB", "CLAUDE_CODE_ARTIFACT_CAP_KB"),
            default=50 * 1024,
        )
        self._phase_cap_override = self._resolve_cap(
            cli_override=phase_artifact_cap_bytes,
            env_keys=("CLAWTEAM_PHASE_ARTIFACT_CAP_KB", "OH_PHASE_ARTIFACT_CAP_KB", "CLAUDE_CODE_PHASE_ARTIFACT_CAP_KB"),
            default=500 * 1024,
        )

    @staticmethod
    def _resolve_cap(cli_override: int | None, env_keys: tuple[str, ...], default: int) -> int:
        if cli_override is not None:
            return cli_override * 1024 if cli_override < 1024 else cli_override
        env_val = _env(*env_keys, default="")
        if env_val:
            return int(env_val) * 1024
        return default

    # ─────────────── lifecycle ───────────────

    def start_sprint(self, goal: str, auto_advance: bool = True) -> SprintState:
        with self._lock:
            sprint_id = uuid.uuid4().hex[:8]
            first_phase = self._first_phase_from_registry()
            state = SprintState(
                sprint_id=sprint_id,
                team_name=self.team_name,
                goal=goal,
                current_phase=first_phase,
                phase_history=[],
                artifacts={},
                participants=[],
                pending_question_ids=[],
                auto_advance=auto_advance,
                workspace_branch="main",
                artifact_cap_bytes=self._artifact_cap_override,
                phase_artifact_cap_bytes=self._phase_cap_override,
                status="running",
            )
            save_sprint_state(state)
            self._emit_phase_transition(None, first_phase, state)
            return state

    def advance_phase(self, sprint_id: str) -> tuple[bool, str]:
        with self._lock:
            state = self._load_by_id(sprint_id)
            # Compose gate chain for current_phase per D-23 + force_interactive_phases
            gates = self._build_gate_chain(state)
            for gate in gates:
                ok, reason = gate.check(state)
                if not ok:
                    return False, reason
            # All gates passed — advance
            next_phase = self._next_phase(state.current_phase)
            if next_phase is None:
                state.status = "completed"
                save_sprint_state(state)
                return True, "sprint completed"
            old_phase = state.current_phase
            state.current_phase = next_phase
            state.phase_history.append({"from": old_phase, "to": next_phase, "at": _now_iso()})
            save_sprint_state(state)
            self._emit_phase_transition(old_phase, next_phase, state)
            return True, ""

    def pause(self, sprint_id: str) -> SprintState:
        with self._lock:
            state = self._load_by_id(sprint_id)
            state.status = "paused"
            save_sprint_state(state)
            return state

    def resume(self, sprint_id: str) -> SprintState:
        with self._lock:
            state = self._load_by_id(sprint_id)
            if state.status not in ("paused", "running"):
                raise ValueError(f"Cannot resume sprint in status={state.status}")
            state.status = "running"
            save_sprint_state(state)
            # Re-emit last phase transition from history (CORE-07)
            if state.phase_history:
                last = state.phase_history[-1]
                self._emit_phase_transition(last.get("from"), last.get("to"), state)
            # Restore careful_enabled on subscriber module
            from clawteam.harness.freeze_registry import set_careful_veto_mode
            set_careful_veto_mode(state.careful_enabled)
            return state

    # ─────────────── queries ───────────────

    def list_sprints(self) -> list[SprintState]:
        sprints_dir = get_data_dir() / "teams" / self.team_name / "sprints"
        if not sprints_dir.exists():
            return []
        result = []
        for sd in sorted(sprints_dir.iterdir()):
            state_path = sd / "state.json"
            if state_path.exists():
                try:
                    result.append(load_sprint_state(self.team_name, sd.name))
                except Exception:
                    continue  # skip corrupted
        return result

    def resolve_sprint(self, id_or_prefix: str) -> SprintState:
        sprints_dir = get_data_dir() / "teams" / self.team_name / "sprints"
        if not sprints_dir.exists():
            raise SprintNotFoundError(id_or_prefix)
        candidates = [p.name for p in sprints_dir.iterdir() if p.name.startswith(id_or_prefix)]
        if not candidates:
            raise SprintNotFoundError(id_or_prefix)
        if len(candidates) > 1:
            raise AmbiguousSprintError(id_or_prefix, candidates)
        return load_sprint_state(self.team_name, candidates[0])

    # ─────────────── internals ───────────────

    def _load_by_id(self, sprint_id: str) -> SprintState:
        try:
            return load_sprint_state(self.team_name, sprint_id)
        except FileNotFoundError:
            # try prefix resolution
            return self.resolve_sprint(sprint_id)

    def _first_phase_from_registry(self) -> str:
        try:
            from clawteam.harness.phase_registry import get_phase_registry
            reg = get_phase_registry()
            phases = reg.phases()
            if phases:
                return phases[0]
        except Exception:
            pass
        return DEFAULT_FIRST_PHASE

    def _next_phase(self, current: str) -> str | None:
        try:
            from clawteam.harness.phase_registry import get_phase_registry
            reg = get_phase_registry()
            phases = reg.phases()
            if current in phases:
                idx = phases.index(current)
                if idx + 1 < len(phases):
                    return phases[idx + 1]
                return None
        except Exception:
            pass
        # default 7-phase gstack order if registry empty
        order = ["think", "plan", "build", "review", "test", "ship", "reflect"]
        if current in order:
            idx = order.index(current)
            return order[idx + 1] if idx + 1 < len(order) else None
        return None

    def _build_gate_chain(self, state: SprintState) -> list:
        """Composition: EvidenceGate → forced_progress_gate → (conditional InteractionGate)."""
        from clawteam.harness.evidence_gate import EvidenceGate
        from clawteam.harness.forced_progress_gate import forced_progress_gate
        from clawteam.harness.interaction_gate import InteractionGate
        gates: list = []
        # EvidenceGate — phase-specific artifact list is populated by GstackSprintPlugin (Phase 3).
        # Phase 2 ships a permissive default that passes on empty artifact list.
        gates.append(EvidenceGate(artifact_names=state.artifacts.keys() and list(state.artifacts.keys()) or []))
        gates.append(forced_progress_gate())
        # InteractionGate insertion logic per D-23 + force_interactive_phases override
        phase = state.current_phase
        has_pending_questions = bool(state.pending_question_ids)
        forced = phase in self.force_interactive_phases
        if forced or (not state.auto_advance) or (state.auto_advance and has_pending_questions):
            gates.append(InteractionGate(sprint_id=state.sprint_id))
        return gates

    def _emit_phase_transition(self, from_phase: str | None, to_phase: str, state: SprintState) -> None:
        try:
            from clawteam.events.types import PhaseTransition
            self.bus.emit(PhaseTransition(
                team_name=self.team_name,
                from_phase=from_phase or "",
                to_phase=to_phase,
                artifacts=list(state.artifacts.keys()),
            ))
        except Exception:
            pass  # never crash sprint on event-emit failure

    return datetime.now(timezone.utc).isoformat()
```
