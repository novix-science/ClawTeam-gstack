pre-deliver hook in `Transport.deliver()`. These are the two hook sites that enforce
the Phase 2 envelope protocol and break A↔B deadlock loops.

Critical invariants:
- **Cycle detector extends existing state, does NOT add a parallel tracker** (specifics
  lesson #1). The 50-entry `recentEvents` window already exists; we read its tail 20
  and check for round-trip counts ≥ 3.
- **Envelope enforcement is scope-limited** at Transport.deliver() to avoid breaking
  the Phase 0 regression matrix. Rule: a message with *any* of persona/step_label/done
  must have *all three* (valid envelope); a message with *none* of them passes through
  (existing templates' mailbox messages, system notifications). This lets gstack sprint
  agents opt in per turn via their role prompt (Phase 3) while existing templates
  continue to work.
- **Sync emit() for CycleDetected / DriftRegression** (Pitfall #7) — suppression state
  must be on disk before decide() returns.

    ```python
    def _topic_hash(self, envelope: RuntimeEnvelope) -> str:
        """D-19 priority chain: dedupe_key → request_id → sha1(content[:128])."""
        if envelope.dedupe_key:
            return envelope.dedupe_key
        req_id = None
        payload = getattr(envelope, "payload", None) or {}
        if isinstance(payload, dict):
            req_id = payload.get("request_id") or payload.get("requestId")
        if req_id:
            return str(req_id)
        content = ""
        if isinstance(payload, dict):
            content = payload.get("content") or ""
        content = content or (envelope.summary or "")
        return hashlib.sha1(content[:128].encode("utf-8")).hexdigest()[:16]


    def _detect_cycle(self, state, envelope: RuntimeEnvelope, topic_hash: str) -> dict | None:
        """Return cycle-hit dict when 3+ round-trips with matching topic_hash in last-20 window."""
        window = state.get("recentEvents", [])[-20:]
        route_a_b = self._route_key(envelope.source, envelope.target)
        route_b_a = self._route_key(envelope.target, envelope.source)
        count_ab = 0
        count_ba = 0
        progress_signal = False
        for e in window:
            if e.get("routeKey") == route_a_b and e.get("topicHash") == topic_hash:
                count_ab += 1
                if e.get("progressSignal"):
                    progress_signal = True
            elif e.get("routeKey") == route_b_a and e.get("topicHash") == topic_hash:
                count_ba += 1
                if e.get("progressSignal"):
                    progress_signal = True
        if progress_signal:
            # Pitfall #3: legitimate iteration with progress — don't fire.
            return None
        if count_ab >= 3 and count_ba >= 3:
            return {
                "topic_hash": topic_hash,
                "route_keys": [route_a_b, route_b_a],
                "reason": (
                    f"3+ round-trips in 20-msg window "
                    f"(A→B={count_ab}, B→A={count_ba})"
                ),
            }
        return None
    ```

    ```python
    """Transport abstract base + Phase 2 pre-deliver envelope validation hook."""

    from __future__ import annotations

    import json
    from abc import ABC, abstractmethod
    from threading import Lock

    from clawteam.events.global_bus import get_event_bus
    from clawteam.events.types import DriftRegression, MalformedEnvelope
    from clawteam.team.envelope import MalformedEnvelopeError, TurnEnvelope
    from clawteam.team.models import TeamMessage


    # Per-agent consecutive-malformed counter (§02-CONTEXT D-09). Module-level state so
    # it survives across transport instances within one process; cleared on SprintConductor
    # teardown (Plan 02-11).
    _MALFORMED_COUNTERS: dict[str, int] = {}
    _COUNTER_LOCK = Lock()
    _DRIFT_THRESHOLD = 8  # §02-CONTEXT D-09


    def _pre_deliver_hooks(data: bytes | str) -> TeamMessage:
        """Phase 2 pre-deliver envelope validation + per-agent drift accounting.

        Pitfall #8 BC invariant: messages with NONE of persona/step_label/done pass
        through without envelope validation. Only messages that carry AT LEAST ONE of
        the three fields must carry ALL THREE (valid envelope). This preserves the
        existing software-dev / hedge-fund / code-review / harness-default /
        research-paper / strategy-room templates while enforcing the envelope for
        gstack-sprint agent turns.

        Returns the parsed TeamMessage on success.
        Raises MalformedEnvelopeError when partial envelope or invalid values.
        """
        raw = data.decode("utf-8") if isinstance(data, bytes) else data
        payload = json.loads(raw)
        msg = TeamMessage.model_validate(payload)

        persona = msg.persona
        step_label = msg.step_label
        done = msg.done
        any_envelope = persona is not None or step_label is not None or done is not None
        if not any_envelope:
            return msg  # BC pass-through

        agent = msg.from_agent or ""
        try:
            TurnEnvelope.model_validate({
                "persona": persona or "",
                "step_label": step_label or "",
                "done": done if done is not None else None,
            })
        except MalformedEnvelopeError as exc:
            with _COUNTER_LOCK:
                count = _MALFORMED_COUNTERS.get(agent, 0) + 1
                _MALFORMED_COUNTERS[agent] = count
            try:
                get_event_bus().emit(MalformedEnvelope(
                    agent=agent, violation=str(exc), turn_id=msg.request_id or "",
                ))
            except Exception:
                pass
            if count >= _DRIFT_THRESHOLD:
                try:
                    get_event_bus().emit(DriftRegression(
                        agent=agent, consecutive_count=count, last_violation=str(exc),
                    ))
                except Exception:
                    pass
            raise
        else:
            # Valid envelope → reset counter (§02-CONTEXT D-09 "counter resets on success").
            with _COUNTER_LOCK:
                _MALFORMED_COUNTERS.pop(agent, None)
            return msg


    def _reset_drift_counters() -> None:
        """Test-only: clear the per-agent malformed-counter state."""
        with _COUNTER_LOCK:
            _MALFORMED_COUNTERS.clear()


    class Transport(ABC):
        @abstractmethod
        def deliver(self, recipient: str, data: bytes) -> None: ...
    ```
